sbs_create_succubus_interaction = {
	icon = sbs_create_succubus_interaction
	desc = sbs_create_succubus_interaction.desc
	category = interaction_category_diplomacy
	use_diplomatic_range = yes
	common_interaction = yes
	cooldown = { years = 25 }
	is_shown = {
		NOT = { has_global_variable = seal_the_gates_of_hell }
		scope:recipient = { has_trait = sbs_lesser_succubus }
		scope:actor = { has_trait = sbs_demon }
	}
	is_valid = {
		scope:actor = {
			NOT = { has_character_modifier = sbs_seal }
		}
		scope:recipient = {
			has_trait_xp = {
				trait = sbs_lesser_succubus
				value >= 100
			}
		}
	}
	send_options_exclusive = no
	populate_recipient_list = {
		scope:recipient = {
			is_ruler = yes
			every_courtier = {
				limit = {
					OR = { NOT = { this = scope:recipient } }
				}
				add_to_list = characters
			}
		}
		scope:actor = {
			every_courtier = {
				limit = {
					scope:recipient = { is_ruler = no }
					NOT = { this = scope:recipient }
				}
				add_to_list = characters
			}
		}
	}
	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:favor = yes }
				add_hook = {
					target = scope:recipient
					type = favor_hook
				}
			}
			add_character_modifier = { modifier = sbs_seal years = 5 }
		}
		scope:recipient = {
			if = {
				limit = { has_trait = saint }
				remove_trait = saint
				set_character_faith_with_conversion = scope:actor.faith
				give_nickname = nick_the_betrayer
			}
			if = {
				limit = { has_trait = immortal }
				remove_trait = immortal
			}
			remove_trait = sbs_lesser_succubus
			add_trait = sbs_succubus
			if = {
				limit = { scope:secondary_recipient = { is_close_family_of = scope:recipient } }
				add_trait_xp = { trait = sbs_succubus track = corruption value = 50 }
			}
			if = {
				limit = {
					scope:hook = yes
					has_hook = scope:actor
				}
				use_hook = scope:actor
			}
			if = {
				limit = { scope:gold = yes }
				pay_short_term_gold = { target = scope:actor gold = scope:recipient.deal_price_value }
			}
			if = {
				limit = { scope:demand_conversion = yes }
				set_character_faith_with_conversion = scope:actor.faith
			}
		}
		scope:secondary_recipient = {
			death = {
				death_reason = death_execution
				killer = scope:recipient
			}
		}
	}
	ai_accept = {
		base = 20
		opinion_modifier = {
			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.5
		}
		opinion_modifier = {
			opinion_target = scope:secondary_recipient
			who = scope:recipient
			multiplier = -0.5
		}
		modifier = { add = 10 scope:actor.faith = scope:recipient.faith }
		modifier = { add = 10 scope:actor.culture = scope:recipient.culture }
		modifier = { add = -5 scope:hook = yes }
		modifier = { add = -5 scope:gold = yes }
		modifier = { add = -5 scope:favor = yes }
		modifier = { add = -10 scope:demand_conversion = yes scope:recipient = { has_trait = zealous } }
		modifier = { add = -10 scope:demand_conversion = yes scope:recipient = { primary_title.tier = tier_kingdom } }
		modifier = { add = -20 scope:demand_conversion = yes scope:recipient = { primary_title.tier = tier_empire } }
		modifier = { add = -30 scope:demand_conversion = yes scope:recipient = { primary_title.tier = tier_hegemony } }
	}
	ai_frequency = 12
	ai_targets = {
		ai_recipients = family
		ai_recipients = courtiers
		ai_recipients = scripted_relations
		ai_recipients = neighboring_rulers
		ai_recipients = nearby_domicile_owners
		max = 5
	}
	ai_potential = { always = yes }
	ai_will_do = {
		base = 0
		modifier = { add = 10 scope:recipient = { is_close_or_extended_family_of = scope:actor } }
		modifier = { add = 100 scope:recipient = { is_child_of = scope:actor } }
	}
}
