become_demon = {

	selection_tooltip = become_demon.t
	decision_group_type = decisions

	is_shown = {
		NOT = { has_global_variable = seal_the_gates_of_hell }
		OR = {
			has_trait = sbs_lesser_succubus
			has_trait = sbs_lesser_demon
			AND = {
				faith.religion = religion:demonic_religion
				faith.religious_head = root
			}
		}
	}
	is_valid = {
		OR = {
			AND = {
				has_trait_xp = { trait = sbs_lesser_succubus track = daring value >= 100 }
				has_trait_xp = { trait = sbs_lesser_succubus track = power value >= 100 }
				has_trait_xp = { trait = sbs_lesser_succubus track = corruption value >= 100 }
				has_trait_xp = { trait = sbs_lesser_succubus track = loyalty value >= 100 }

				has_trait_xp = { trait = sbs_lesser_demon value >= 100 }
				has_trait = education_intrigue_5
				has_trait = education_diplomacy_5
				has_trait = education_stewardship_5
				has_trait = education_martial_5
				has_trait = education_learning_5
			}
		}
	}
	effect = {
		if = {
			limit = { faith = faith:nature faith.religious_head = root }
			if = {
				limit = { NOT = { dynasty = dynasty:888888 } }
				random_living_character = {
					limit = { faith = faith:nature is_ruler = no }
					copy_inheritable_appearance_from = character:nature_character

						base = 0
						modifier = { is_lowborn = no add = 1 }
						modifier = { dynasty = dynasty:888888 add = 100 }
						modifier = { dynasty = dynasty:888888 is_female = yes is_adult = yes add = 1000 }
						modifier = { dynasty = dynasty:888888 has_trait = sbs_demon is_adult = yes add = 10000 }
					}
				}
				random_held_title = {
					limit = { this = title:k_nature }
					create_title_and_vassal_change = {
						type = usurped
						save_scope_as = change
						add_claim_on_loss = no
					}
					change_title_holder_include_vassals = {
						holder = scope:demon_child
						change = scope:change
						take_baronies = no
					}
					resolve_title_and_vassal_change = scope:change
				}
			}
		}
		else_if = {
			limit = { faith = faith:pleasure faith.religious_head = root }
			if = {
				limit = { NOT = { dynasty = dynasty:666666 } }
				random_living_character = {

						modifier = { dynasty = dynasty:666666 add = 100 }
						modifier = { dynasty = dynasty:666666 is_female = yes is_adult = yes add = 1000 }

						type = usurped

alienate_people = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds" }
	desc = alienate_people.desc
	confirm_text = alienate_people.c
	selection_tooltip = alienate_people.t
	decision_group_type = decisions

	is_shown = {
		is_ruler = yes
		is_landed = yes
		OR = {
			NOT = { capital_province.culture = culture }
			NOT = { capital_province.faith = faith }
			any_vassal = {
				primary_title.tier >= tier_county
				NOR = {
					faith = root.faith
					capital_province.faith = root.faith
					culture = root.culture
					capital_province.culture = root.culture
				}
			}
		}
		OR = {
			has_trait = sbs_demon
			has_trait = sbs_succubus
		}
	}
	effect = {
		if = {
			limit = { NOT = { capital_province.faith = root.faith } }
			capital_province.county = {
				set_county_faith = root.faith
			}
		}
		if = {
			limit = { NOT = { capital_province.culture = root.culture } }
			capital_province.county = {
				set_county_culture = root.culture
			}
		}
	}
	ai_potential = { always = yes }
	ai_check_interval = 12
	ai_will_do = { base = 100 }
}
