namespace = carn_slave

# 0001. You have been enslaved
carn_slave.0001 = {
	type = character_event
	title = carn_slave.0001.t
	desc = carn_slave.0001.desc
	theme = death
	left_portrait = {
		character = root
		animation = prisonhouse
	}
	override_background = {
		reference = gallows
	}
	immediate = {
		play_music_cue = mx_cue_stress
	}
	option = {
		name = carn_slave.0001.a
	}
}

# 0002. Slaves are freed when they gain land for any reason
carn_slave.0002 = {
	hidden = yes
	trigger = {
		has_trait = slave
	}
	immediate = {
		carn_free_slave_effect = yes
	}
}

# 0003. Slave newborns get their owner relation
carn_slave.0003 = {
	hidden = yes

	trigger = {
		scope:child = {
			has_trait = slave
		}
	}

	immediate = {
		scope:mother = {
			if = {
				limit = { any_relation = { type = slave_owner } }
				random_relation = {
					type = slave_owner
					save_scope_as = owner
				}
			}
			else_if = {
				limit = { exists = liege }
				liege = { save_scope_as = owner }
			}
			else = {
				location.province_owner = { save_scope_as = owner }
			}
			scope:owner = {
				set_relation_slave = scope:child
				scope:child = {
					set_relation_slave_owner = prev
				}
				add_courtier = scope:child
			}
			scope:child = {
				#set_to_lowborn = yes
			}
		}
	}
}

# 0004. Return slave to owner's court if slave left
carn_slave.0004 = {
	hidden = yes

	trigger = {
		is_ruler = no
		has_trait = slave
		has_relation_slave_owner = scope:old_employer
		is_alive = yes
	}

	immediate = {
		scope:old_employer = { add_courtier = root }
	}
}

carn_slave.0005 = {
	hidden = yes

	trigger = {
		is_ruler = no
		any_relation = { type = slave }
		is_alive = yes
	}

	immediate = {
		save_scope_as = seller
		every_relation = {
			type = slave
			if = {
				limit = { exists = liege }
				liege = { save_scope_as = buyer }
			}
			else = {
				random_ruler = { save_scope_as = buyer }
			}
			save_scope_as = slave
			carn_buy_slave_effect = { SLAVE_PRICE = 0 }
		}
	}
}

carn_slave.0006 = {
	hidden = yes

	trigger = {
		has_trait = slave
		is_female = yes
		is_adult = yes
		fertility >= 0.1
		any_relation = { type = slave_owner }
	}

	immediate = {
		random_living_character = {
			limit = {
				has_trait = slave
				this != scope:slave
				is_female = no
				NOT = { is_close_or_extended_family_of = scope:slave }
			}
			weight = {
				base = 1
				modifier = { is_at_same_location = scope:slave add = 1000 }
				modifier = { in_diplomatic_range = scope:slave_owner add = 1000 }
			}
			save_scope_as = breeder
		}
		if = {
			limit = { exists = scope:breeder }
			had_sex_with_effect = { CHARACTER = scope:breeder PREGNANCY_CHANCE = pregnancy_chance }
		}
	}
}
