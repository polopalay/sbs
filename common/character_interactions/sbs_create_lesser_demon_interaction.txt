sbs_create_lesser_demon_interaction = {
	icon = sbs_create_lesser_demon_interaction
	desc = sbs_create_lesser_demon_interaction.desc
	category = interaction_category_diplomacy
	use_diplomatic_range = yes
	cooldown = { years = 10 }
	common_interaction = yes
	is_shown = {
		NOT = { has_global_variable = seal_the_gates_of_hell }
		scope:recipient = { NOT = { has_trait_with_flag = sbs_demon_trait } }
		scope:actor = { OR = { has_trait = sbs_true_demon has_trait = sbs_demon has_trait = sbs_succubus } }
	}
	is_valid = {
		scope:recipient.age >= 16
		scope:actor = {
			trigger_if = {
				limit = { scope:recipient = { is_female = yes } }
				sbs_has_brothel_3_trigger = yes
			}
			trigger_if = {
				limit = { scope:recipient = { is_female = no } }
				sbs_has_alchemy_3_trigger = yes
			}
		}
	}
	localization_values = {
		EXTORTIONATE_RANSOM_COST = scope:recipient.increased_ransom_cost_value
		RANSOM_COST = scope:recipient.increased_ransom_cost_value
	}
	send_option = {
		is_shown = {
			scope:actor != scope:recipient
			scope:actor = {
				dynasty ?= { has_dynasty_perk = fp1_pillage_legacy_3 }
			}
			scope:recipient ?= { gold >= scope:recipient.increased_ransom_cost_value }
		}
		flag = extortionate_gold
		localization = "RANSOM_EXTORTIONATE_GOLD_OPTION"
	}
	send_option = {
		is_shown = {
			scope:actor != scope:recipient
			scope:actor = {
				dynasty ?= { has_dynasty_perk = fp1_pillage_legacy_3 }
			}
			scope:recipient ?= {
				gold < scope:recipient.increased_ransom_cost_value
				gold >= 1
			}
		}
		flag = extortionate_current_gold
		localization = "RANSOM_EXTORTIONATE_CURRENT_GOLD_OPTION"
	}
	send_option = {
		is_shown = {
			scope:actor != scope:recipient
			scope:actor = {
				OR = {
					is_lowborn = yes
					NOT = { dynasty = { has_dynasty_perk = fp1_pillage_legacy_3 } }
				}
			}
			scope:recipient ?= { gold >= scope:recipient.increased_ransom_cost_value }
		}
		flag = gold
		localization = "RANSOM_GOLD_OPTION"
	}
	send_option = {
		is_shown = {
			scope:actor != scope:recipient
			scope:actor = {
				OR = {
					is_lowborn = yes
					NOT = { dynasty = { has_dynasty_perk = fp1_pillage_legacy_3 } }
				}
			}
			scope:recipient ?= {
				gold < scope:recipient.increased_ransom_cost_value
				gold >= 1
			}
		}
		flag = current_gold
		localization = "RANSOM_CURRENT_GOLD_OPTION"
	}
	send_option = {
		is_shown = { scope:actor != scope:recipient }
		is_valid = { scope:recipient = { has_usable_hook = scope:actor } }
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	send_option = {
		is_shown = { scope:actor != scope:recipient }
		is_valid = {
			scope:actor = {
				can_add_hook = {
					target = scope:recipient
					type = favor_hook
				}
			}
		}
		flag = favor
		localization = "RANSOM_FAVOR_OPTION"
	}
	send_option = {
		flag = demand_conversion
		localization = "RELEASE_DEMAND_CONVERSION"
		is_shown = { scope:recipient.faith != scope:actor.faith }
		is_valid = {
			custom_description = {
				text = cannot_demand_religious_head_conversion
				NAND = {
					exists = scope:recipient.faith.religious_head
					scope:recipient.faith.religious_head = scope:recipient
				}
			}
			custom_description = {
				text = cannot_demand_holy_order_master_conversion
				NAND = {
					exists = scope:recipient.faith
					scope:recipient.faith = { any_faith_holy_order = { leader = scope:recipient } }
				}
			}
		}
	}
	send_option = {
		flag = gain_hook
		localization = "RELEASE_GAIN_HOOK"
		is_shown = {
			trigger_if = {
				limit = { scope:actor = { is_ai = yes } }
				scope:recipient = {
					OR = {
						is_vassal_of = scope:actor
						liege ?= scope:actor
					}
				}
			}
		}
		is_valid = {
			scope:actor = {
				can_add_hook = {
					type = favor_hook
					target = scope:recipient
				}
			}
		}
	}
	send_option = {
		flag = recruit
		localization = "RELEASE_RECRUIT"
		is_shown = {
			trigger_if = {
				limit = { scope:actor = { is_ai = yes } }
				scope:recipient = {
					OR = {
						has_relation_best_friend = scope:actor
						has_relation_friend = scope:actor
						has_relation_lover = scope:actor
						has_relation_soulmate = scope:actor
						has_secret_relation_lover = scope:actor
						is_heir_of = scope:actor
					}
				}
			}
			scope:recipient = {
				is_ruler = no
				NOT = { is_courtier_of = scope:actor }
				OR = {
					is_adult = yes
					AND = {
						is_adult = no
						OR = {
							trigger_if = { limit = { exists = liege } NOT = { is_heir_of = liege } }
							NOT = { exists = liege }
						}
					}
				}
			}
		}
	}
	send_option = {
		flag = demand_admin
		localization = "RELEASE_DEMAND_ADMIN"
		is_shown = {
			scope:actor = { government_allows = administrative }
			scope:actor = scope:recipient.liege
			scope:recipient = {
				highest_held_title_tier >= tier_county
				is_landed = yes
				is_ruler = yes
				NOT = { government_allows = administrative }
				NOT = { government_has_flag = government_is_tribal }
				NOT = { scope:recipient ?= scope:recipient.faith.religious_head }
				NOT = { scope:recipient ?= scope:actor.diarch }
			}
		}
		is_valid = { scope:recipient = { NOT = { government_allows = administrative } } }
	}
	send_options_exclusive = no
	on_accept = {
		scope:actor = {
			if = {
				limit = { OR = { scope:favor = yes scope:gain_hook = yes } }
				add_hook = { target = scope:recipient type = favor_hook }
			}
		}
		scope:recipient = {
			if = {
				limit = { OR = { scope:gold = yes scope:extortionate_gold = yes } }
				pay_short_term_gold = { target = scope:actor gold = scope:recipient.increased_ransom_cost_value }
			}
			if = {
				limit = { OR = { scope:current_gold = yes scope:extortionate_current_gold = yes } }
				pay_short_term_gold = { target = scope:actor gold = this.gold }
			}
			if = {
				limit = { scope:hook = yes has_hook = scope:actor }
				use_hook = scope:actor
			}
			if = {
				limit = { scope:demand_conversion = yes }
				set_character_faith = scope:actor.faith
			}
			if = {
				limit = { scope:recruit = yes }
				if = {
					limit = { scope:actor = { culture = { has_cultural_parameter = can_recruit_prisoners_easily } } }
					scope:actor = {
						if = {
							limit = { can_add_hook = { target = scope:recipient type = loyalty_hook } }
							add_hook = { type = loyalty_hook target = scope:recipient }
						}
					}
				}
				scope:actor = { add_courtier = scope:recipient }
			}
			if = {
				limit = { scope:demand_admin = yes }
				change_to_administrative_effect = yes
			}
			if = {
				limit = { is_female = yes }
				add_trait = sbs_lesser_succubus
				sbs_mind_break_effect = yes
			}
			else = {
				add_trait = sbs_lesser_demon
				add_character_modifier = { modifier = sbs_buff years = 100 }
			}
			if = {
				limit = { has_trait = saint }
				remove_trait = saint
				give_nickname = nick_the_betrayer
			}
			if = {
				limit = { has_trait = chaste }
				remove_trait = chaste
				add_trait = lustful
			}
			if = {
				limit = { has_trait = immortal }
				remove_trait = immortal
				if = {
					limit = { is_female = yes }
					add_trait_xp = { trait = sbs_lesser_succubus track = corruption value = 100 }
					add_trait_xp = { trait = sbs_lesser_succubus track = power value = 100 }
				}
				else = {
					add_trait_xp = { trait = sbs_lesser_demon value = 100 }
				}
			}
		}
	}
	ai_accept = {
		base = 20
		opinion_modifier = { opinion_target = scope:actor who = scope:recipient multiplier = 0.25 }
		modifier = { add = -5 OR = { scope:gold = yes scope:extortionate_gold = yes } }
		modifier = { add = -3 OR = { scope:current_gold = yes scope:extortionate_current_gold = yes } }
		modifier = { add = -3 scope:hook = yes }
		modifier = { add = -3 scope:favor = yes }
		modifier = { add = -3 scope:gain_hook = yes }
		modifier = { add = -5 scope:demand_conversion = yes }
		modifier = { add = -2 scope:recruit = yes }
		modifier = { add = -4 scope:demand_admin = yes }
		modifier = { add = 10 scope:actor.faith = scope:recipient.faith }
		modifier = { add = 10 scope:actor.culture = scope:recipient.culture }
		modifier = { add = -20 scope:demand_conversion = yes scope:recipient = { has_trait = zealous } }
		modifier = { add = -20 scope:demand_conversion = yes scope:recipient.primary_title.tier = tier_kingdom }
		modifier = { add = -40 scope:demand_conversion = yes scope:recipient.primary_title.tier = tier_empire }
		modifier = { add = -80 scope:demand_conversion = yes scope:recipient.primary_title.tier = tier_hegemony }
	}
	ai_frequency = 12
	ai_targets = { ai_recipients = family max = 10 }
	ai_targets = {
		ai_recipients = scripted_relations
		ai_recipients = neighboring_rulers
		ai_recipients = realm_characters
		max = 10
	}
	ai_potential = { always = yes }
	ai_will_do = {
		base = 0

		modifier = {
			factor = 0
			scope:actor != scope:recipient
			NOR = {
				scope:extortionate_gold = yes
				scope:extortionate_current_gold = yes
				scope:gold = yes
				scope:current_gold = yes
				scope:hook = yes
				scope:favor = yes
				scope:gain_hook = yes
				scope:demand_conversion = yes
				scope:recruit = yes
				scope:demand_admin = yes
			}
		}

		modifier = { add = 25 scope:recipient = { is_child_of = scope:actor } }
		modifier = { add = 10 scope:recipient = { is_close_or_extended_family_of = scope:actor } }
		modifier = { add = 10 scope:recipient.dynasty = scope:actor.dynasty }
	}
}
