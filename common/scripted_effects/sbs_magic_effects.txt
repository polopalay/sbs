sbs_steal_body = {
	$VICTIM$ = { save_scope_as = victim }
	$STEALER$ = { save_scope_as = stealer }

	if = {
		limit = {
			OR = {
				AND = {
					scope:victim = { is_female = yes }
					scope:stealer = { is_female = yes }
				}
				AND = {
					scope:victim = { is_male = yes }
					scope:stealer = { is_male = yes }
				}
			}
		}

		if = {
			limit = { scope:stealer = { is_female = yes } }
			create_character = {
				faith = scope:victim.faith
				culture = scope:victim.culture
				age = 18
				gender = female
				random_traits = no
				after_creation = {
					clear_traits = yes
					copy_traits = scope:victim
				}
				save_scope_as = victim_traits
			}
		}

		else = {
			create_character = {
				faith = scope:victim.faith
				culture = scope:victim.culture
				age = 18
				gender = male
				random_traits = no
				after_creation = {
					clear_traits = yes
					copy_traits = scope:victim
				}
				save_scope_as = victim_traits
			}
		}

		scope:victim = {
			save_scope_as = demon
			clear_traits = yes
			copy_traits = scope:stealer

			if = {
				limit = { is_immortal = yes }
				set_immortal_age = 18
			}

			if = {
				limit = { has_trait = pregnant }
				remove_trait = pregnant
			}
		}

		scope:stealer = {
			clear_traits = yes
			copy_traits = scope:victim_traits

			if = {
				limit = { is_immortal = yes }
				set_immortal_age = 18
			}

			if = {
				limit = { has_trait = pregnant }
				remove_trait = pregnant
			}
		}

		scope:victim_traits = {
			death = { death_reason = death_vanished }
		}
	}

	else = {
		if = {
			limit = {
				scope:stealer = { is_female = yes }
			}

			create_character = {
				faith = scope:victim.faith
				culture = scope:victim.culture
				father = scope:victim.father
				mother = scope:victim.mother
				age = scope:victim.age
				gender = female
				random_traits = no

				after_creation = {
					change_first_name = {
						template_character = scope:victim
					}

					set_house = scope:victim.house
					copy_inheritable_appearance_from = scope:victim

					clear_traits = yes
					copy_traits = scope:stealer

					if = {
						limit = { is_immortal = yes }
						set_immortal_age = 18
					}

					if = {
						limit = { has_trait = pregnant }
						remove_trait = pregnant
					}
				}

				save_scope_as = demon
			}
		}

		else = {
			create_character = {
				faith = scope:victim.faith
				culture = scope:victim.culture
				father = scope:victim.father
				mother = scope:victim.mother
				age = scope:victim.age
				gender = male
				random_traits = no

				after_creation = {
					change_first_name = {
						template_character = scope:victim
					}

					set_house = scope:victim.house
					copy_inheritable_appearance_from = scope:victim

					clear_traits = yes
					copy_traits = scope:stealer

					if = {
						limit = { is_immortal = yes }
						set_immortal_age = 18
					}

					if = {
						limit = { has_trait = pregnant }
						remove_trait = pregnant
					}
				}

				save_scope_as = demon
			}
		}

		scope:victim = {
			every_vassal = {
				add_to_list = sbs_steal_body_vassals
			}

			create_title_and_vassal_change = {
				type = usurped
				save_scope_as = title_change
				add_claim_on_loss = no
			}

			every_held_title = {
				change_title_holder_include_vassals = {
					holder = scope:demon
					change = scope:title_change
					take_baronies = no
				}
			}

			resolve_title_and_vassal_change = scope:title_change
		}

		create_title_and_vassal_change = {
			type = granted
			save_scope_as = vassal_change
			add_claim_on_loss = no
		}

		every_in_list = {
			list = sbs_steal_body_vassals

			limit = {
				highest_held_title_tier < scope:demon.highest_held_title_tier
				NOT = {
					liege = scope:demon
				}
			}

			change_liege = {
				liege = scope:demon
				change = scope:vassal_change
			}
		}

		resolve_title_and_vassal_change = scope:vassal_change

		scope:stealer = {
			clear_traits = yes
			copy_traits = scope:victim

			if = {
				limit = { is_immortal = yes }
				set_immortal_age = 18
			}

			if = {
				limit = { has_trait = pregnant }
				remove_trait = pregnant
			}
		}

		scope:victim = {
			death = { death_reason = death_vanished }
		}
	}

	scope:stealer = {
		set_player_character = scope:demon
	}
}

sbs_remove_one_bad_habit_effect = {

			OR = {
				this = trait:drunkard
				this = trait:hashishiyah
				this = trait:rakish
				this = trait:flagellant
				this = trait:profligate
				this = trait:comfort_eater
				this = trait:inappetetic
				this = trait:reclusive
				this = trait:irritable
			}
		}
		save_scope_as = sbs_bad_habit_trait
	}

		limit = { exists = scope:sbs_bad_habit_trait }
		remove_trait = scope:sbs_bad_habit_trait
	}
}

sbs_replace_one_sin_effect = {
	save_scope_as = sbs_sinful_character
	random_character_trait = {
		limit = {
			has_trait_category = personality
			scope:sbs_sinful_character.faith = {
				trait_is_sin = prev
			}
		}
		save_scope_as = sbs_sinful_personality_trait
	}
	if = {
		limit = { exists = scope:sbs_sinful_personality_trait }
		remove_trait = scope:sbs_sinful_personality_trait
		random_trait = {
			limit = {
				has_trait_category = personality
				save_temporary_scope_as = sbs_candidate_virtue
				scope:sbs_sinful_character.faith = { trait_is_virtue = prev }
				NOR = {
					scope:sbs_sinful_character = { has_trait = scope:sbs_candidate_virtue }
					any_opposite_trait = {
						scope:sbs_sinful_character = { has_trait = scope:sbs_candidate_virtue }
					}
				}
			}
			save_scope_as = sbs_virtuous_personality_trait
		}
		if = {
			limit = { exists = scope:sbs_virtuous_personality_trait }
			add_trait_force_tooltip = scope:sbs_virtuous_personality_trait
		}
	}
}
