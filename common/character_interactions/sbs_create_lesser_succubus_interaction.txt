sbs_create_lesser_succubus_interaction = {
	icon = sbs_create_lesser_succubus_interaction
	desc = sbs_create_lesser_succubus_interaction.desc
	category = interaction_category_diplomacy
	use_diplomatic_range = yes
	cooldown = { years = 25 }
	common_interaction = yes
	is_shown = {
		NOT = { has_global_variable = seal_the_gates_of_hell }
		scope:recipient = {
			is_female = yes
				has_trait = saint
		}
		scope:actor = {
			OR = {
				has_trait = sbs_demon
				AND = {
					has_trait = sbs_succubus
					is_valid_construction_alchemy3_trigger = yes
				}
			}
		}
	}
	is_valid = {
		scope:actor.faith.religion = religion:demonic_religion
		scope:recipient = {
			OR = {
				has_trait = deviant
				any_secret = { secret_type = secret_deviant }
			}
			OR = {
				has_trait = witch
				any_secret = { secret_type = secret_witch }
			}
		}
	}
	send_option = {
		is_shown = {
			scope:recipient = { is_ruler = yes }
		}
		flag = small_price
		localization = "small_price"
	}
	send_options_exclusive = no
	on_accept = {
		scope:recipient = {
			if = {
				limit = { has_trait = saint }
				remove_trait = saint
				set_character_faith_with_conversion = scope:actor.faith
				give_nickname = nick_the_betrayer
			}
			if = {
				limit = { has_trait = immortal }
				remove_trait = immortal
			}
			add_trait = sbs_lesser_succubus
			if = {
				limit = { scope:small_price = yes }
				add_opinion = { target = scope:actor modifier = carn_wants_to_be_your_slave_opinion }
				every_vassal = {
					limit = { primary_title.tier >= tier_county }
					save_scope_as = recipient
					scope:actor = { save_scope_as = actor }
					offer_vassalization_interaction_effect = yes
				}
				every_held_title = {
					create_title_and_vassal_change = {
						type = usurped
						save_scope_as = change
						add_claim_on_loss = no
					}
					change_title_holder_include_vassals = {
						holder = scope:actor
						change = scope:change
						take_baronies = no
					}
					resolve_title_and_vassal_change = scope:change
				}
				carn_enslave_effect = { SLAVE = this OWNER = scope:actor DRAMA = no }
			}
		}
	}
	auto_accept = {
		scope:recipient = {
			is_courtier_of = scope:actor
		}
	}
	ai_accept = {
		base = 20
		opinion_modifier = {
			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.5
		}
		modifier = { add = -50 scope:small_price = yes }
	}
	ai_frequency = 12
	ai_targets = {
		ai_recipients = family
		ai_recipients = scripted_relations
		ai_recipients = neighboring_rulers
		ai_recipients = nearby_domicile_owners
		max = 5
	}
	ai_potential = { always = yes }
	ai_will_do = {
		base = 10
		modifier = { add = 100 scope:recipient = { is_child_of = scope:actor } }
		modifier = { add = 5 scope:recipient = { is_female = yes is_courtier_of = scope:actor } }
		modifier = { add = 5 scope:recipient = { is_female = yes is_vassal_of = scope:actor } }
		modifier = { add = 100 scope:recipient = { is_ai = no } }
		modifier = { add = 100 scope:recipient = { has_trait = saint } }
	}
}
