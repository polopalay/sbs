# Seize Slave
carn_seize_slave_interaction = {
	interface_priority = 100
	common_interaction = yes
	category = interaction_category_prison

	desc = carn_seize_slave_interaction_desc

	ai_min_reply_days = 0
	ai_max_reply_days = 0

	ai_targets = {
		ai_recipients = prisoners
	}

	ai_frequency = 12

	is_shown = {
		scope:recipient = {
			has_trait = slave
			is_imprisoned_by = scope:actor
			NOT = { has_relation_slave_owner = scope:actor }
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			NOT = { has_strong_hook = scope:actor }
			custom_description = {
				text = "currently_being_tortured"
				NOT = { has_character_flag = is_being_tortured }
			}
		}
	}

	auto_accept = {
		always = yes
	}

	on_accept = {
		scope:recipient = {
			every_relation = {
				type = slave_owner

				hidden_effect = {
					remove_relation_slave = prev
				}
			}
			set_relation_slave_owner = scope:actor
		}
		scope:actor = {
			set_relation_slave = scope:recipient
		}
	}

	ai_will_do = {
		base = 0

		opinion_modifier = {
			opinion_target = scope:recipient.liege
			multiplier = -1
		}

		ai_value_modifier = {
			ai_compassion = -1
		}

		ai_value_modifier = {
			ai_honor = -1
		}
	}
}
