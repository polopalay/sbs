sbs_ask_for_become_lesser_succubus_interaction = {
	icon = sbs_ask_for_become_lesser_succubus_interaction
	desc = sbs_ask_for_become_lesser_succubus_interaction.desc
	category = interaction_category_diplomacy
	use_diplomatic_range = {
		scope:actor = { is_ai = yes }
	}
	common_interaction = yes
	cooldown = { years = 10 }

	is_shown = {
		NOT = { has_global_variable = seal_the_gates_of_hell }
		scope:actor = {
			is_female = yes
			has_trait = saint
			NOR = {
				has_trait = sbs_lesser_succubus
				has_trait = sbs_succubus
				has_trait = sbs_demon
			}
		}
		scope:recipient = {
			OR = {
				has_trait = sbs_demon
				AND = {
					has_trait = sbs_succubus
					is_valid_construction_alchemy3_trigger = yes
				}
			}
		}
	}

	is_valid = {
		scope:actor = {
			OR = {
				primary_title.tier >= tier_kingdom
				AND = {
					culture = scope:recipient.culture
					is_close_or_extended_family_of = scope:recipient
				}
			}
			OR = {
				has_trait = deviant
				any_secret = { secret_type = secret_deviant }
			}
			ON = {
				has_trait = witch
				any_secret = { secret_type = secret_witch }
			}
		}
	}

	localization_values = {
		RANSOM_COST = major_gold_value
	}
	send_option = {
		is_valid = {
			scope:recipient ?= { gold >= major_gold_value }
		}
		flag = gold
		localization = "RANSOM_GOLD_OPTION"
	}
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	send_option = {
		is_valid = {
			exists = scope:recipient
			scope:recipient = {
				can_add_hook = {
					target = scope:actor
					type = favor_hook
				}
			}
		}
		flag = favor
		localization = "RANSOM_OWE_FAVOR_OPTION"
	}
	send_option = {
		is_shown = {
			scope:actor = { is_ruler = yes }
		}
		flag = small_price
		localization = "small_price"
	}
	send_options_exclusive = no

	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:gold = yes }
				pay_short_term_gold = {
					target = scope:recipient
					gold = major_gold_value
				}
			}
			if = {
				limit = {
					scope:hook = yes
					has_hook = scope:recipient
				}
				use_hook = scope:recipient
			}
			if = {
				limit = {
					scope:demand_conversion = yes
				}
				set_character_faith_with_conversion = scope:recipient.faith
			}
			if = {
				limit = { has_trait = saint }
				remove_trait = saint
				set_character_faith_with_conversion = scope:recipient.faith
				give_nickname = nick_the_betrayer
			}
			if = {
				limit = { has_trait = immortal }
				remove_trait = immortal
			}
			add_trait = sbs_lesser_succubus
			if = {
				limit = { scope:small_price = yes }
				add_opinion = { target = scope:recipient modifier = carn_wants_to_be_your_slave_opinion }
				every_vassal = {
					limit = { primary_title.tier >= tier_county }
					save_scope_as = recipient
					scope:recipient = { save_scope_as = actor }
					offer_vassalization_interaction_effect = yes
				}
				every_held_title = {
					create_title_and_vassal_change = {
						type = usurped
						save_scope_as = change
						add_claim_on_loss = no
					}
					change_title_holder_include_vassals = {
						holder = scope:recipient
						change = scope:change
						take_baronies = no
					}
					resolve_title_and_vassal_change = scope:change
				}
				carn_enslave_effect = { SLAVE = this OWNER = scope:recipient DRAMA = no }
			}
		}
		scope:recipient = {
			if = {
				limit = { scope:favor = yes }
				add_hook = {
					target = scope:actor
					type = fabrication_hook
				}
			}
		}
	}
	ai_accept = {
		base = -25
		opinion_modifier = {
			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.5
		}
		modifier = { add = 5 scope:gold = yes }
		modifier = { add = 5 scope:hook = yes }
		modifier = { add = 5 scope:favor = yes }
		modifier = { scope:recipient = { is_courtier_of = scope:actor } add = 10 }
		modifier = { add = 50 scope:small_price = yes }
	}

	ai_potential = { always = yes }
	ai_frequency = 12
	ai_targets = {
		ai_recipients = family
		ai_recipients = head_of_faith
		max = 5
	}

	ai_will_do = {
		base = 50
	}
}
