become_demon = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds" }
	desc = become_demon.desc
	confirm_text = become_demon.c
	selection_tooltip = become_demon.t
	decision_group_type = decisions

	is_shown = {
		NOT = { has_trait = sbs_true_demon }
		NOT = { has_global_variable = seal_the_gates_of_hell }
		OR = {
			has_trait = sbs_lesser_succubus
			has_trait = sbs_lesser_demon
		}
	}
	is_valid = {
		OR = {
			AND = {
				has_trait_xp = { trait = sbs_lesser_succubus track = corruption value >= 100 }
				has_trait_xp = { trait = sbs_lesser_succubus track = loyalty value >= 100 }
				age >= 100
			}
			AND = {
				has_trait_xp = { trait = sbs_lesser_demon value >= 100 }
				age >= 100
			}
		}
	}
	effect = {
		if = {
			limit = { is_female = yes }
			add_trait = sbs_succubus
			set_immortal_age = 18
			remove_trait = sbs_lesser_succubus
		}
		else = {
			add_trait = sbs_demon
			set_immortal_age = 18
			remove_trait = sbs_lesser_demon
		}
	}
	ai_potential = { always = yes }
	ai_check_interval = 12
	ai_will_do = { base = 100 }
}

seal_the_gates_of_hell = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds" }
	desc = seal_the_gates_of_hell.desc
	confirm_text = seal_the_gates_of_hell.c
	selection_tooltip = seal_the_gates_of_hell.t
	decision_group_type = decisions

	is_shown = {
		is_ruler = yes
		has_trait = saint
		NOT = { has_global_variable = seal_the_gates_of_hell }
	}
	is_valid = {
		province:2065.religion = religion:christianity_religion
		province:5965.religion = religion:christianity_religion
		province:482.religion = religion:christianity_religion
		province:2575.religion = religion:christianity_religion
		province:4582.religion = religion:christianity_religion
		province:6038.religion = religion:christianity_religion
		custom_description = { text = nature_is_seal has_global_variable = seal_nature }
		custom_description = { text = pleasure_is_seal has_global_variable = seal_pleasure }
		custom_description = { text = violence_is_seal has_global_variable = seal_violence }
		custom_description = { text = greedy_is_seal has_global_variable = seal_greedy }
	}
	effect = {
		faith:nature.religious_head = { death = { death_reason = death_execution killer = root } }
		faith:pleasure.religious_head = { death = { death_reason = death_execution killer = root } }
		faith:violence.religious_head = { death = { death_reason = death_execution killer = root } }
		faith:greedy.religious_head = { death = { death_reason = death_execution killer = root } }

		destroy_title = title:k_nature
		destroy_title = title:k_pleasure
		destroy_title = title:k_violence
		destroy_title = title:k_greedy

		custom_description = {
			text = seal_the_gates_of_hell
			set_global_variable = { name = seal_the_gates_of_hell value = yes }
		}
	}
	ai_potential = { always = yes }
	ai_check_interval = 12
	ai_will_do = { base = 100 }
}
