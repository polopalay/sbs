sbs_devils_gift_interaction = {
	desc = sbs_devils_gift_interaction.desc
	category = interaction_category_hostile
	use_diplomatic_range = yes
	cooldown = { years = 1 }
	cooldown_against_recipient = { years = 2 }
	common_interaction = yes

	is_shown = {
		scope:actor = { has_trait_with_flag = sbs_demon_trait }
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			OR = {
				has_trait = sbs_true_demon
				has_trait = sbs_demon
				has_trait = sbs_succubus
				has_trait_xp = { trait = sbs_lesser_succubus track = power value >= 25 }
				has_trait_xp = { trait = sbs_lesser_demon value >= 50 }
			}
		}
	}

	send_option = {
		is_shown = {
			scope:actor = {
				has_trait = sbs_true_demon
			}
			scope:recipient = {
				is_ruler = no
			}
		}
		flag = game_concept_ruler
		localization = game_concept_ruler
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				NOR = {
					any_secret = { secret_type = secret_deviant }
					has_trait = deviant
				}
			}
		}
		flag = deviant
		localization = trait_deviant
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				NOR = {
					any_secret = { secret_type = secret_witch }
					has_trait = witch
				}
			}
		}
		flag = witch
		localization = trait_witch
	}

	send_option = {
		is_shown = { scope:recipient = { has_character_modifier = sbs_seal } }
		flag = sbs_seal
		localization = sbs_seal
	}

	send_option = {
		flag = sbs_buff
		localization = sbs_buff
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				NOT = { has_character_modifier = possessed_by_lust_demon_modifier }
			}
		}
		flag = possessed_by_lust_demon_modifier
		localization = possessed_by_lust_demon_modifier
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				NOT = { has_character_modifier = legend_bloodthirsty_warrior }
			}
		}
		flag = legend_bloodthirsty_warrior
		localization = legend_bloodthirsty_warrior
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				NOT = { has_character_modifier = learned_in_demonology }
			}
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_1_trigger = yes
			}
		}
		flag = learned_in_demonology
		localization = learned_in_demonology
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				NOT = { has_character_modifier = possessed_prophetic_visions }
			}
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_1_trigger = yes
			}
		}
		flag = possessed_prophetic_visions
		localization = possessed_prophetic_visions
	}

	send_option = {
		is_valid = {
			scope:actor = { sbs_has_alchemy_3_trigger = yes }
		}
		flag = intellect_good
		localization = trait_intellect_good
	}

	send_option = {
		is_shown = {
			scope:recipient = { diplomacy_lifestyle_perk_points >= 1 }
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_2_trigger = yes
			}
		}
		flag = convert_diplomacy_perk
		localization = diplomacy
	}

	send_option = {
		is_shown = {
			scope:recipient = { martial_lifestyle_perk_points >= 1 }
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_2_trigger = yes
			}
		}
		flag = convert_martial_perk
		localization = martial
	}

	send_option = {
		is_shown = {
			scope:recipient = { stewardship_lifestyle_perk_points >= 1 }
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_2_trigger = yes
			}
		}
		flag = convert_stewardship_perk
		localization = stewardship
	}

	send_option = {
		is_shown = {
			scope:recipient = { intrigue_lifestyle_perk_points >= 1 }
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_2_trigger = yes
			}
		}
		flag = convert_intrigue_perk
		localization = intrigue
	}

	send_option = {
		is_shown = {
			scope:recipient = { learning_lifestyle_perk_points >= 1 }
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_2_trigger = yes
			}
		}
		flag = convert_learning_perk
		localization = learning
	}

	send_option = {
		is_shown = {
			scope:actor = { is_valid_construction_alchemy_trigger = yes }
		}
		flag = reset_perks
		localization = REFUND_PERKS
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				is_adult = yes
				has_trait_rank = {
					trait = education_learning
					rank < 5
				}
			}
			scope:actor = {
				faith.religion = religion:demonic_religion
			}
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_3_trigger = yes
			}
		}
		flag = education_learning
		localization = trait_education_learning
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				is_adult = yes
				has_trait_rank = {
					trait = education_martial
					rank < 5
				}
			}
			scope:actor = {
				faith.religion = religion:demonic_religion
			}
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_3_trigger = yes
			}
		}
		flag = education_martial
		localization = trait_education_martial
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				is_adult = yes
				has_trait_rank = {
					trait = education_intrigue
					rank < 5
				}
			}
			scope:actor = {
				faith.religion = religion:demonic_religion
			}
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_3_trigger = yes
			}
		}
		flag = education_intrigue
		localization = trait_education_intrigue
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				is_adult = yes
				has_trait_rank = {
					trait = education_diplomacy
					rank < 5
				}
			}
			scope:actor = {
				faith.religion = religion:demonic_religion
			}
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_3_trigger = yes
			}
		}
		flag = education_diplomacy
		localization = trait_education_diplomacy
	}

	send_option = {
		is_shown = {
			scope:recipient = {
				is_adult = yes
				has_trait_rank = {
					trait = education_stewardship
					rank < 5
				}
			}
			scope:actor = {
				faith.religion = religion:demonic_religion
			}
		}
		is_valid = {
			scope:actor = {
				sbs_has_alchemy_3_trigger = yes
			}
		}
		flag = education_stewardship
		localization = trait_education_stewardship
	}

	send_options_exclusive = yes

	on_accept = {
		scope:actor = {
			if = {
				limit = { has_trait = sbs_lesser_succubus }
				add_trait_xp = {
					trait = sbs_lesser_succubus
					track = power
					value = -25
				}
			}
			else_if = {
				limit = { has_trait = sbs_lesser_demon }
				add_trait_xp = {
					trait = sbs_lesser_demon
					value = -50
				}
			}
		}

		scope:recipient = {
			if = {
				limit = { scope:game_concept_ruler = yes }
				create_landless_adventurer_title_effect = {
					REASON = flag:voluntary
					FLAVOR_CHAR = scope:recipient
				}
			}
			else_if = {
				limit = { scope:sbs_buff = yes }
				random_list = {
					5 = {
						add_character_modifier = {
							modifier = sbs_generals
							years = 50
						}
					}
					10 = {
						modifier = {
							is_valid_construction_alchemy_trigger = yes
							add = 20
						}
						add_character_modifier = {
							modifier = sbs_buff
							years = 100
						}
					}
				}
			}
			else_if = {
				limit = { scope:sbs_seal = yes }
				remove_character_modifier = sbs_seal
			}
			else_if = {
				limit = { scope:deviant = yes }
				if = {
					limit = { trait_is_shunned_or_criminal_in_faith_or_liege_trigger = { TRAIT = trait:deviant } }
					add_secret = { type = secret_deviant }
				}
				else = {
					add_trait = deviant
				}
			}
			else_if = {
				limit = { scope:witch = yes }
				if = {
					limit = { trait_is_shunned_or_criminal_in_faith_or_liege_trigger = { TRAIT = trait:witch } }
					add_secret = { type = secret_witch }
				}
				else = {
					add_trait = witch
				}
			}
			else_if = {
				limit = { scope:possessed_by_lust_demon_modifier = yes }
				add_character_modifier = { modifier = possessed_by_lust_demon_modifier years = 10 }
			}
			else_if = {
				limit = { scope:legend_bloodthirsty_warrior = yes }
				add_character_modifier = { modifier = legend_bloodthirsty_warrior years = 10 }
			}
			else_if = {
				limit = { scope:possessed_prophetic_visions = yes }
				add_character_modifier = { modifier = possessed_prophetic_visions years = 10 }
			}
			else_if = {
				limit = { scope:learned_in_demonology = yes }
				add_character_modifier = { modifier = learned_in_demonology years = 10 }
			}
			else_if = {
				limit = { scope:intellect_good = yes }
				if = {
					limit = { has_trait_rank = { trait = intellect_good rank < 3 } }
					change_trait_rank = { trait = intellect_good rank = 1 }
				}
				else = {
					set_trait_rank = { trait = intellect_good rank = 0 }
				}
			}
			else_if = {
				limit = { scope:convert_diplomacy_perk = yes }
				add_diplomacy_lifestyle_perk_points = -1
				add_diplomacy_skill = 1
			}
			else_if = {
				limit = { scope:convert_martial_perk = yes }
				add_martial_lifestyle_perk_points = -1
				add_martial_skill = 1
			}
			else_if = {
				limit = { scope:convert_stewardship_perk = yes }
				add_stewardship_lifestyle_perk_points = -1
				add_stewardship_skill = 1
			}
			else_if = {
				limit = { scope:convert_intrigue_perk = yes }
				add_intrigue_lifestyle_perk_points = -1
				add_intrigue_skill = 1
			}
			else_if = {
				limit = { scope:convert_learning_perk = yes }
				add_learning_lifestyle_perk_points = -1
				add_learning_skill = 1
			}
			else_if = {
				limit = { scope:reset_perks = yes }
				refund_all_perks = yes
				remove_all_landless_inspector_modifiers = yes
				remove_all_perk_lifestyle_traits_effect = yes
			}
			else_if = {
				limit = { scope:education_learning = yes }
				change_trait_rank = { trait = education_learning rank = 1 }
			}
			else_if = {
				limit = { scope:education_martial = yes }
				change_trait_rank = { trait = education_martial rank = 1 }
			}
			else_if = {
				limit = { scope:education_intrigue = yes }
				change_trait_rank = { trait = education_intrigue rank = 1 }
			}
			else_if = {
				limit = { scope:education_diplomacy = yes }
				change_trait_rank = { trait = education_diplomacy rank = 1 }
			}
			else_if = {
				limit = { scope:education_stewardship = yes }
				change_trait_rank = { trait = education_stewardship rank = 1 }
			}
		}
	}

	auto_accept = yes

	ai_frequency = 12

	ai_targets = {
		ai_recipients = self
	}

	ai_targets = {
		ai_recipients = family
		ai_recipients = courtiers
		max = 10
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 1

		modifier = {
			add = 25
			scope:actor = { OR = { has_trait = sbs_true_demon has_trait = sbs_demon has_trait = sbs_succubus } }
			scope:recipient = { has_trait_with_flag = sbs_demon_trait }
			scope:recipient.dynasty = scope:actor.dynasty
			scope:recipient != scope:actor
		}

		modifier = {
			add = 25
			scope:actor = { OR = { has_trait = sbs_lesser_demon has_trait = sbs_lesser_succubus } }
			scope:recipient = scope:actor
		}
	}
}
