sbs_create_lesser_demon_interaction = {
	icon = sbs_create_lesser_demon_interaction
	desc = sbs_create_lesser_demon_interaction.desc
	category = interaction_category_diplomacy
	use_diplomatic_range = yes
	cooldown = { years = 10 }
	common_interaction = yes
	is_shown = {
		NOT = { has_global_variable = seal_the_gates_of_hell }
		scope:recipient = {
			NOT = { has_trait_with_flag = sbs_demon_trait }
		}
		scope:actor = {
			OR = {
				has_trait = sbs_demon
				AND = {
					has_trait = sbs_succubus
					is_valid_construction_alchemy3_trigger = yes
				}
			}
		}
	}
	is_valid = {
		scope:recipient = {
			is_adult = yes
		}
		scope:actor = {
			faith.religion = religion:demonic_religion
			NOT = { has_character_modifier = sbs_debuff }
		}
	}
	localization_values = {
		RANSOM_COST = massive_gold_value
	}
	send_option = {
		is_valid = { scope:recipient ?= { gold >= massive_gold_value } }
		flag = gold
		localization = "RANSOM_GOLD_OPTION"
	}
	send_option = {
		is_valid = {
			scope:recipient = {
				has_usable_hook = scope:actor
			}
		}
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	send_option = {
		is_valid = {
			exists = scope:recipient
			scope:actor = {
				can_add_hook = {
					target = scope:recipient
					type = favor_hook
				}
			}
		}
		flag = favor
		localization = "RANSOM_OWE_FAVOR_OPTION"
	}
	send_option = {
		flag = demand_conversion
		localization = "RELEASE_DEMAND_CONVERSION"
		is_shown = {
			NOT = { scope:actor.faith = scope:recipient.faith }
		}
		is_valid = {
			# Cannot ask Religious Heads to convert
			custom_description = {
				text = cannot_demand_religious_head_conversion
				NOT = {
					AND = {
						exists = scope:recipient.faith.religious_head
						scope:recipient.faith.religious_head = scope:recipient
					}
				}
			}
			# Cannot ask Holy Order Master to convert
			custom_description = {
				text = cannot_demand_holy_order_master_conversion
				NAND = {
					exists = scope:recipient.faith
					scope:recipient.faith = {
						any_faith_holy_order = { leader = scope:actor }
					}
				}
			}
		}
	}
	send_options_exclusive = no
	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:favor = yes }
				add_hook = {
					target = scope:recipient
					type = favor_hook
				}
			}
			add_character_modifier = { modifier = sbs_debuff years = 2 }
		}
		scope:recipient = {
			if = {
				limit = { is_female = yes }
				add_trait = sbs_lesser_succubus
			}
			else = {
				add_trait = sbs_lesser_demon
				add_character_modifier = sbs_buff
			}
			if = {
				limit = { has_trait = saint }
				remove_trait = saint
				give_nickname = nick_the_betrayer
			}
			if = {
				limit = { has_trait = chaste }
				remove_trait = chaste
				add_trait = lustful
			}
			if = {
				limit = { has_trait = immortal }
				remove_trait = immortal
				add_trait_xp = { trait = sbs_lesser_succubus value = 100 }
			}
			if = {
				limit = {
					scope:hook = yes
					has_hook = scope:actor
				}
				use_hook = scope:actor
			}
			if = {
				limit = { scope:gold = yes }
				pay_short_term_gold = { target = scope:actor gold = scope:recipient.deal_price_value }
			}
			if = {
				limit = { scope:demand_conversion = yes }
				set_character_faith_with_conversion = scope:actor.faith
			}
		}
	}
	ai_accept = {
		base = 20
		opinion_modifier = {
			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.25
		}
		modifier = { add = 10 scope:actor.faith = scope:recipient.faith }
		modifier = { add = 10 scope:actor.culture = scope:recipient.culture }
		modifier = { add = -5 scope:hook = yes }
		modifier = { add = -5 scope:gold = yes }
		modifier = { add = -5 scope:favor = yes }
		modifier = { add = -20 scope:demand_conversion = yes scope:recipient = { has_trait = zealous } }
		modifier = { add = -20 scope:demand_conversion = yes scope:recipient.primary_title.tier = tier_kingdom }
		modifier = { add = -40 scope:demand_conversion = yes scope:recipient.primary_title.tier = tier_empire }
		modifier = { add = -80 scope:demand_conversion = yes scope:recipient.primary_title.tier = tier_hegemony }
	}
	ai_frequency = 12
	ai_targets = {
		ai_recipients = family
		ai_recipients = scripted_relations
		ai_recipients = neighboring_rulers
		max = 10
	}
	ai_potential = { always = yes }
	ai_will_do = {
		base = 0
		modifier = { add = 25 scope:recipient = { is_child_of = scope:actor } }
		modifier = { add = 10 scope:recipient = { is_close_or_extended_family_of = scope:actor } }
	}
}
