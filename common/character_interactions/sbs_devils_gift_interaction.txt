sbs_devils_gift_interaction = {
	desc = sbs_devils_gift_interaction.desc
	category = interaction_category_hostile
	use_diplomatic_range = yes
	cooldown = { years = 1 }
	cooldown_against_recipient = { years = 2 }
	common_interaction = yes
	is_shown = {
		scope:actor = {
			has_trait_with_flag = sbs_demon_trait
		}
	}
	is_valid = {
		scope:actor = {
			OR = {
				has_trait = sbs_demon
				has_trait_xp = { trait = sbs_succubus track = daring value >= 10 }
				has_trait_xp = { trait = sbs_lesser_succubus value >= 25 }
				has_trait_xp = { trait = sbs_lesser_demon value >= 50 }
			}
		}
	}
	send_option = {
		is_shown = { scope:actor = { has_trait = sbs_demon dynasty ?= { any_dynasty_member = { count < 13 } } this = scope:recipient } }
		flag = child
		localization = nick_the_child
	}
	send_option = {
		is_shown = { scope:actor = { has_trait = sbs_demon } scope:recipient = { is_ruler = no has_trait_with_flag = sbs_demon_trait } }
		flag = game_concept_ruler
		localization = game_concept_ruler
	}
	send_option = {
		is_shown = { scope:actor = { is_ai = no } scope:actor != scope:recipient }
		flag = dna
		localization = DNA_MODIFIERS
	}
	send_option = {
		is_shown = { scope:recipient = { is_adult = yes has_trait_rank = { trait = education_learning rank < 5 } } }
		flag = education_learning
		localization = trait_education_learning
	}
	send_option = {
		is_shown = { scope:recipient = { is_adult = yes has_trait_rank = { trait = education_martial rank < 5 } } }
		flag = education_martial
		localization = trait_education_martial
	}
	send_option = {
		is_shown = { scope:recipient = { is_adult = yes has_trait_rank = { trait = education_intrigue rank < 5 } } }
		flag = education_intrigue
		localization = trait_education_intrigue
	}
	send_option = {
		is_shown = { scope:recipient = { is_adult = yes has_trait_rank = { trait = education_diplomacy rank < 5 } } }
		flag = education_diplomacy
		localization = trait_education_diplomacy
	}
	send_option = {
		is_shown = { scope:recipient = { is_adult = yes has_trait_rank = { trait = education_stewardship rank < 5 } } }
		flag = education_stewardship
		localization = trait_education_stewardship
	}
	send_option = {
		is_shown = { scope:recipient = { has_character_modifier = sbs_seal } }
		flag = sbs_seal
		localization = sbs_seal
	}
	send_option = {
		flag = sbs_buff
		localization = sbs_buff
	}
	send_option = {
		flag = sbs_generals
		localization = sbs_generals
	}
	send_options_exclusive = yes
	on_accept = {
		scope:actor = {
			if = {
				limit = { has_trait = sbs_succubus }
				add_trait_xp = { trait = sbs_succubus track = daring value = -10 }
			}
			else_if = {
				limit = { has_trait = sbs_lesser_succubus }
				add_trait_xp = { trait = sbs_lesser_succubus value = -25 }
			}
			else_if = {
				limit = { has_trait = sbs_lesser_demon }
				add_trait_xp = { trait = sbs_lesser_demon value = -50 }
			}
		}
		scope:recipient = {
			random_list = {
				10 = 0
				0 = {
					modifier = { scope:child = yes add = 10 }
					if = {
						limit = { is_female = yes }
						save_scope_as = mother
						random_living_character = {
							limit = { has_trait = sbs_demon is_female = no is_adult = yes }
							save_scope_as = father
						}
					}
					else = {
						save_scope_as = father
						random_living_character = {
							limit = { has_trait = sbs_demon is_female = yes is_adult = yes }
							save_scope_as = mother
						}
					}
					create_character = {
						age = 0
						employer = scope:actor
						location = scope:actor.location
						mother = scope:mother
						father = scope:father
						dynasty_house = scope:actor.house
						culture = scope:actor.culture
						faith = scope:actor.faith
						gender_female_chance = 50
						liege = scope:actor
					}
				}
				0 = {
					modifier = { scope:game_concept_ruler = yes add = 10 }
					create_landless_adventurer_title_effect = {
						REASON = flag:voluntary
						FLAVOR_CHAR = scope:recipient
					}
				}
				0 = {
					modifier = { scope:dna = yes add = 40 }
					scope:actor = { copy_inheritable_appearance_from = scope:recipient }
				}
				0 = {
					modifier = { scope:education_learning = yes add = 20 }
					change_trait_rank = { trait = education_learning rank = 1 }
				}
				0 = {
					modifier = { scope:education_martial = yes add = 20 }
					change_trait_rank = { trait = education_martial rank = 1 }
				}
				0 = {
					modifier = { scope:education_intrigue = yes add = 20 }
					change_trait_rank = { trait = education_intrigue rank = 1 }
				}
				0 = {
					modifier = { scope:education_diplomacy = yes add = 20 }
					change_trait_rank = { trait = education_diplomacy rank = 1 }
				}
				0 = {
					modifier = { scope:education_stewardship = yes add = 20 }
					change_trait_rank = { trait = education_stewardship rank = 1 }
				}
				0 = {
					modifier = { scope:sbs_seal = yes add = 40 }
					remove_character_modifier = sbs_seal
				}
				0 = {
					modifier = { scope:sbs_buff = yes add = 30 }
					add_character_modifier = sbs_buff
				}
				0 = {
					modifier = { scope:sbs_generals = yes add = 30 }
					add_character_modifier = { modifier = sbs_generals years = 5 }
				}
			}
		}
	}
	auto_accept = yes
	ai_frequency = 6
	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = family
		ai_recipients = courtiers
		max = 20
	}
	ai_potential = { always = yes }
	ai_will_do = {
		base = 0
		modifier = { add = 25 scope:actor = { has_trait = sbs_demon } scope:recipient = { is_close_family_of = scope:actor } }
		modifier = { add = 50 scope:actor = { has_trait = sbs_demon } scope:recipient = { is_child_of = scope:actor } }
		modifier = { add = 50 scope:recipient = scope:actor }
	}
}
