sbs_absorb_interaction = {
	desc = sbs_absorb_interaction.desc
	category = interaction_category_vassal
	use_diplomatic_range = yes
	common_interaction = yes
	cooldown = { years = 1 }
	cooldown_against_recipient = { years = 10 }
	is_shown = {
		scope:actor = {
			OR = {
				has_trait = sbs_succubus
				has_trait = sbs_lesser_succubus
				has_trait = sbs_seed_of_demon
				AND = { has_trait = sbs_seed_of_demon NOT = { this = scope:recipient } }
			}
		}
	}
	is_valid = {
		scope:recipient = {
			OR = {
				has_character_modifier = pussy_cum
				has_character_modifier = ass_cum
				has_character_modifier = face_cum
				has_character_modifier = body_cum
				has_character_modifier = sbs_buff
				has_character_modifier = sbs_generals
				has_trait = sbs_demon
				scope:actor = { has_trait = sbs_seed_of_demon }
			}
		}
	}
	on_accept = {
		scope:actor = {
			if = {
				limit = { has_trait = sbs_lesser_succubus }
				if = {
					limit = { scope:recipient = { has_character_modifier = pussy_cum } }
					add_trait_xp = { trait = sbs_lesser_succubus value = 10 }
				}
				if = {
					limit = { scope:recipient = { has_character_modifier = ass_cum } }
					add_trait_xp = { trait = sbs_lesser_succubus value = 10 }
				}
				if = {
					limit = { scope:recipient = { has_character_modifier = face_cum } }
					add_trait_xp = { trait = sbs_lesser_succubus value = 10 }
				}
				if = {
					limit = { scope:recipient = { has_character_modifier = body_cum } }
					add_trait_xp = { trait = sbs_lesser_succubus value = 10 }
				}
				if = {
					limit = { scope:recipient = { has_character_modifier = sbs_buff } }
					add_trait_xp = { trait = sbs_seed_of_demon value = 20 }
				}
				if = {
					limit = { scope:recipient = { has_character_modifier = sbs_generals } }
					add_trait_xp = { trait = sbs_seed_of_demon value = 50 }
				}
				if = {
					limit = { scope:recipient = { has_trait = sbs_demon } }
					add_trait_xp = { trait = sbs_succubus track = loyalty value = 50 }
				}
			}
			else_if = {
				limit = { has_trait = sbs_succubus }
				if = {
					limit = { scope:recipient = { has_character_modifier = pussy_cum } }
					add_trait_xp = { trait = sbs_succubus track = power value = 5 }
				}
				if = {
					limit = { scope:recipient = { has_character_modifier = ass_cum } }
					add_trait_xp = { trait = sbs_succubus track = power value = 5 }
				}
				if = {
					limit = { scope:recipient = { has_character_modifier = body_cum } }
					add_trait_xp = { trait = sbs_succubus track = daring value = 5 }
				}
				if = {
					limit = { scope:recipient = { has_character_modifier = face_cum } }
					add_trait_xp = { trait = sbs_succubus track = daring value = 5 }
				}
				if = {
					limit = { scope:recipient = { has_character_modifier = sbs_buff } }
					add_trait_xp = { trait = sbs_succubus track = corruption value = 1 }
				}
				if = {
					limit = { scope:recipient = { has_character_modifier = sbs_generals } }
					add_trait_xp = { trait = sbs_succubus track = loyalty value = 2 }
				}
				if = {
					limit = { scope:recipient = { has_trait = sbs_demon } }
					add_trait_xp = { trait = sbs_succubus track = corruption value = 10 }
					add_trait_xp = { trait = sbs_succubus track = loyalty value = 10 }
				}
			}
			else_if = {
				limit = { has_trait = seed_of_demon }
				add_trait_xp = { trait = sbs_seed_of_demon value = 20 }
				if = {
					limit = { scope:recipient = { has_character_modifier = sbs_buff } }
					add_trait_xp = { trait = sbs_seed_of_demon value = 20 }
				}
				if = {
					limit = { scope:recipient = { has_character_modifier = sbs_generals } }
					add_trait_xp = { trait = sbs_seed_of_demon value = 50 }
				}
				if = {
					limit = { scope:recipient = { has_trait = sbs_demon } }
					add_trait_xp = { trait = sbs_succubus track = loyalty value = 50 }
				}
			}
		}
		scope:recipient = {
			if = {
				limit = { actor = { has_trait = seed_of_demon } }
				add_character_modifier = { modifier = sbs_debuff years = 10 }
			}
			else = {
				remove_character_modifier = face_cum
				remove_character_modifier = body_cum
				remove_character_modifier = ass_cum
				remove_character_modifier = pussy_cum
			}
		}
	}
	auto_accept = {
		scope:recipient = {
			OR = {
				is_courtier_of = scope:actor
				has_relation_slave_owner = scope:actor
			}
		}
	}
	ai_accept = {
		base = -50
		opinion_modifier = {
			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.25
		}
		modifier = { add = 20 scope:recipient.faith = scope:actor.faith }
		modifier = { add = 40 scope:recipient = { is_parent_of = scope:actor } }
		modifier = { add = 10 scope:recipient = { has_dread_level_towards = { target = scope:actor level = 1 } } desc = INTIMIDATED_REASON }
		modifier = { add = 20 scope:recipient = { has_dread_level_towards = { target = scope:actor level = 2 } } desc = COWED_REASON }
	}
	ai_frequency = 12
	ai_targets = {
		ai_recipients = self
		ai_recipients = head_of_faith
		ai_recipients = family
		ai_recipients = courtiers
		ai_recipients = vassals
	}
	ai_potential = { always = yes }
	ai_will_do = { base = 100 }
}
