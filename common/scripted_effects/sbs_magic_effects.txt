sbs_steal_body = {
	$VICTIM$ = { save_scope_as = victim }
	$STEALER$ = { save_scope_as = stealer }
	if = {
		limit = { scope:victim = { is_female = yes } }
		scope:victim = {
			save_scope_as = demon
		}
	}
	else = {
		create_character = {
			faith = scope:victim.faith
			culture = scope:victim.culture
			father = scope:victim.father
			mother = scope:victim.mother
			age = scope:victim.age
			gender = female
			random_traits = no
			after_creation = {
				change_first_name = { template_character = scope:victim }
				set_house = scope:victim.house
				copy_inheritable_appearance_from = scope:victim
				clear_traits = yes
				copy_traits = scope:stealer
				if = {
					limit = { has_trait = pregnant }
					remove_trait = pregnant
				}
			}
			save_scope_as = demon
		}
		scope:victim = {
			every_held_title = {
				create_title_and_vassal_change = {
					type = usurped
					save_scope_as = change
					add_claim_on_loss = no
				}
				change_title_holder_include_vassals = {
					holder = scope:demon
					change = scope:change
					take_baronies = no
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
	}

	scope:stealer = {
		set_player_character = scope:demon
		clear_traits = yes
		copy_traits = scope:victim
		if = {
			limit = { has_trait = pregnant }
			remove_trait = pregnant
		}
	}
	scope:victim = {
		if = {
			limit = { is_female = no }
			death = { death_reason = death_vanished }	
		}
		else = {
			clear_traits = yes
			copy_traits = scope:demon
			if = {
				limit = { has_trait = pregnant }
				remove_trait = pregnant
			}
		}
	}
}
