			}
			if = {
				limit = { scope:current_gold = yes }
				pay_short_term_gold = { target = scope:recipient gold = scope:actor.current_gold_value }
			}

				limit = { scope:hook = yes has_hook = scope:recipient }
				use_hook = scope:recipient
			}
			save_scope_as = buyer
		}
		scope:recipient = {
			if = {
				limit = { scope:favor = yes }
				add_hook = { target = scope:actor type = favor_hook }
			}
			save_scope_as = seller
		}
