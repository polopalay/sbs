	construction_time = slow_construction_time

	can_construct_potential = {
		scope:holder = {
			OR = {
				has_trait_with_flag = sbs_demon_trait

			}
		}

	show_disabled = yes

		years_of_fertility = 25

	ai_value = {

		modifier = {

			}
		}
		modifier = {
			add = 1000
