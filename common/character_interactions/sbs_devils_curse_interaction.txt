sbs_devils_curse_interaction = {
	desc = sbs_devils_curse_interaction.desc
	category = interaction_category_hostile
	use_diplomatic_range = yes
	cooldown = { years = 1 }
	common_interaction = yes
	is_shown = {
		scope:recipient != scope:actor
		scope:actor = {
			is_adult = yes
			has_trait_with_flag = sbs_demon_trait
		}
		scope:recipient = {
			is_adult = yes
			NOT = {
				has_trait = saint
				has_trait_with_flag = sbs_demon_trait
			}
		}
	}
	is_valid = {
		scope:actor = {
			OR = {
				has_trait = sbs_demon
				has_trait_xp = { trait = sbs_succubus track = daring value >= 25 }
				has_trait_xp = { trait = sbs_lesser_demon value >= 50 }
				has_trait_xp = { trait = sbs_lesser_succubus value >= 100 }
			}
		}
	}
	send_option = {
		is_shown = { scope:actor = { NOT = { has_character_modifier = stanky_modifier } } }
		flag = stanky_modifier
		localization = stanky_modifier
	}
	send_option = {
		is_shown = { scope:actor = { NOT = { has_character_modifier = possessed_by_lust_demon_modifier } } }
		flag = possessed_by_lust_demon_modifier
		localization = possessed_by_lust_demon_modifier
	}
	send_option = {
		is_shown = { scope:recipient = { NOT = { has_trait = pneumonic } } }
		flag = pneumonic
		localization = trait_pneumonic
	}
	send_option = {
		is_shown = { scope:recipient = { NOT = { has_trait = typhus } } }
		flag = typhus
		localization = trait_typhus
	}
	send_option = {
		is_shown = { scope:recipient = { NOT = { has_trait = consumption } } }
		flag = consumption
		localization = trait_consumption
	}
	send_option = {
		is_shown = { scope:recipient = { NOT = { has_trait = bubonic_plague } } }
		flag = bubonic_plague
		localization = trait_bubonic_plague
	}
	send_option = {
		is_shown = { scope:recipient = { NOT = { has_trait = measles } } }
		flag = measles
		localization = trait_measles
	}
	send_option = {
		is_shown = { scope:recipient = { NOT = { has_trait = dysentery } } }
		flag = dysentery
		localization = trait_dysentery
	}
	send_option = {
		is_shown = { scope:recipient = { NOT = { has_trait = ergotism } } }
		flag = ergotism
		localization = trait_ergotism
	}
	send_option = {
		is_shown = { scope:recipient = {  NOT = { has_character_modifier = sbs_seal } is_lowborn = yes is_ruler = no } }
		flag = sbs_seal
		localization = sbs_seal
	}
	send_option = {
		is_shown = { scope:recipient = { is_ai = yes } }
		flag = MAPMODE_PLAYERS_LABEL
		localization = MAPMODE_PLAYERS_LABEL
	}
	send_options_exclusive = no

	on_accept = {
		scope:actor = {
			if = {
				limit = { has_trait = sbs_succubus }
				add_trait_xp = { trait = sbs_succubus track = daring value = -25 }
			}
			else_if = {
				limit = { has_trait = sbs_lesser_demon }
				add_trait_xp = { trait = sbs_lesser_demon value = -50 }
			}
			else_if = {
				limit = { has_trait = sbs_lesser_succubus }
				add_trait_xp = { trait = sbs_lesser_succubus value = -100 }
			}
		}
		scope:recipient = {
			random_list = {
				5 = 0
				0 = {
					modifier = { scope:actor = { is_ai = yes } NOT = { has_character_modifier = stanky_modifier } add = 5 }
					modifier = { scope:stanky_modifier = yes add = 100 }
					add_character_modifier = stanky_modifier
				}
				0 = {
					modifier = { scope:actor = { is_ai = yes } NOT = { has_character_modifier = possessed_by_lust_demon_modifier } add = 5 }
					modifier = { scope:possessed_by_lust_demon_modifier = yes add = 100 }
					add_character_modifier = possessed_by_lust_demon_modifier
				}
				0 = {
					modifier = { scope:pneumonic = yes add = 100 }
					add_trait = pneumonic
				}
				0 = {
					modifier = { scope:typhus = yes add = 100 }
					add_trait = typhus
				}
				0 = {
					modifier = { scope:consumption = yes add = 100 }
					add_trait = consumption
				}
				0 = {
					modifier = { scope:bubonic_plague = yes add = 100 }
					add_trait = bubonic_plague
				}
				0 = {
					modifier = { scope:measles = yes add = 100 }
					add_trait = measles
				}
				0 = {
					modifier = { scope:dysentery = yes add = 100 }
					add_trait = dysentery
				}
				0 = {
					modifier = { scope:ergotism = yes add = 100 }
					add_trait = ergotism
				}
				0 = {
					modifier = { scope:actor = { is_ai = yes } has_trait = slave NOT = { has_character_modifier = sbs_seal } add = 100 }
					modifier = { scope:sbs_seal = yes add = 100 }
					add_character_modifier = sbs_seal
				}
				0 = {
					modifier = { scope:MAPMODE_PLAYERS_LABEL = yes add = 100 }
					sbs_steal_body = { VICTIM = scope:recipient STEALER = scope:actor }	
				}
				1 = {
					scope:actor = { add_character_modifier = sbs_buff }
					add_character_modifier = { modifier = sbs_debuff years = 10 }
				}
			}
		}
	}
	auto_accept = yes
	ai_frequency = 6
	ai_targets = {
		ai_recipients = liege
		ai_recipients = vassals
		ai_recipients = peer_vassals
		ai_recipients = scripted_relations
		ai_recipients = neighboring_rulers
		ai_recipients = nearby_domicile_owners
		ai_recipients = courtiers
		max = 5
	}
	ai_potential = { always = yes }
	ai_will_do = {
		base = 0
		modifier = { add = 100 scope:actor = { has_trait = sbs_lesser_demon has_relation_slave = scope:recipient } }
	}
}
