carn_on_slave_enslaved = {
	effect = {
		random_relation = {
			type = slave_owner
			save_scope_as = new_owner
		}
		create_character_memory = {
			type = carnx_enslaved_memory
			participants = {
				enslaver = scope:new_owner
			}
		}
	}
}

carn_on_slave_freed = {
	effect = {
		create_character_memory = {
			type = carnx_freed_memory
			participants = {
				emancipator = scope:former_owner
			}
		}
	}
}
