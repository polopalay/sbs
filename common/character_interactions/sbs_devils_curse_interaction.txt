sbs_devils_curse_interaction = {
	desc = sbs_devils_curse_interaction.desc
	category = interaction_category_hostile
	use_diplomatic_range = yes
	cooldown = { years = 1 }
	common_interaction = yes
	is_shown = {
		NOT = { scope:recipient = scope:actor }
		scope:actor = {
			OR = {
				has_trait = sbs_demon
				has_trait = sbs_succubus
				has_trait = sbs_seed_of_demon
			}
		}
		scope:recipient = {
			NOT = {
				has_trait = saint
				has_trait_with_flag = sbs_demon_trait
			}
		}
	}
	is_valid = {
		scope:actor = {
			OR = {
				has_trait = sbs_demon
				AND = {
					has_trait_xp = { trait = sbs_succubus track = daring value >= 25 }
					has_trait_xp = { trait = sbs_succubus track = power value >= 25 }
				}
				has_trait_xp = { trait = sbs_seed_of_demon value >= 5 }
			}
		}
	}
	on_accept = {
		scope:actor = {
			if = {
				limit = { has_trait = sbs_succubus }
				add_trait_xp = { trait = sbs_succubus track = daring value = -25 }
				add_trait_xp = { trait = sbs_succubus track = power value = -25 }
			}
			else_if = {
				limit = { has_trait = sbs_seed_of_demon }
				add_trait_xp = { trait = sbs_seed_of_demon value = -5 }
			}
		}
		scope:recipient = {
			random_list = {
				2 = {
					modifier = { has_trait = pneumonic add = -2 }
					modifier = { scope:actor = { has_trait = sbs_seed_of_demon } add = -1 }
					add_trait = pneumonic
				}
				2 = {
					modifier = { has_trait = typhus add = -2 }
					modifier = { scope:actor = { has_trait = sbs_seed_of_demon } add = -1 }
					add_trait = typhus
				}
				2 = {
					modifier = { has_trait = consumption add = -2 }
					modifier = { scope:actor = { has_trait = sbs_seed_of_demon } add = -1 }
					add_trait = consumption
				}
				2 = {
					modifier = { has_trait = bubonic_plague add = -2 }
					modifier = { scope:actor = { has_trait = sbs_seed_of_demon } add = -1 }
					add_trait = bubonic_plague
				}
				2 = {
					modifier = { has_trait = measles add = -2 }
					modifier = { scope:actor = { has_trait = sbs_seed_of_demon } add = -1 }
					add_trait = measles
				}
				2 = {
					modifier = { has_trait = dysentery add = -2 }
					modifier = { scope:actor = { has_trait = sbs_seed_of_demon } add = -1 }
					add_trait = dysentery
				}
				2 = {
					modifier = { has_trait = ergotism add = -2 }
					modifier = { scope:actor = { has_trait = sbs_seed_of_demon } add = -1 }
					add_trait = ergotism
				}
				10 = {
					modifier = { has_character_modifier = stanky_modifier add = -10 }
					add_character_modifier = modifier = stanky_modifier
				}
				10 = {
					modifier = { has_character_modifier = stanky_modifier add = -10 }
					add_character_modifier = modifier = stanky_modifier
				}
				10 = {
					modifier = { has_character_modifier = possessed_by_lust_demon_modifier add = -10 }
					add_character_modifier = modifier = possessed_by_lust_demon_modifier
				}
				2 = {
					add_character_modifier = { modifier = sbs_debuff years = 10 }
					scope:actor = { add_character_modifier = sbs_buff }
				}
				1 = {
					add_opinion = { target = scope:actor modifier = carn_wants_to_be_your_slave_opinion }
				}
				1 = {
					modifier = { has_character_modifier = sbs_seal add = -100 }
					modifier = { scope:actor = { has_trait = sbs_seed_of_demon } scope:recipient = { has_trait = slave } add = 99 }
					add_character_modifier = sbs_seal
				}
			}
		}
	}
	auto_accept = yes
	ai_frequency = 6
	ai_targets = {
		ai_recipients = liege
		ai_recipients = vassals
		ai_recipients = peer_vassals
		ai_recipients = scripted_relations
		ai_recipients = neighboring_rulers
		ai_recipients = nearby_domicile_owners
		ai_recipients = courtiers
		max = 5
	}
	ai_potential = { always = yes }
	ai_will_do = {
		base = 0
		modifier = { add = 50 scope:recipient = { has_relation_nemesis = scope:actor } }
		modifier = { add = 25 scope:recipient = { has_relation_rival = scope:actor } }
		modifier = { add = -25 scope:recipient = { has_relation_friend = scope:actor } }
		modifier = { add = -50 scope:recipient = { has_relation_lover = scope:actor } }
		modifier = { add = -50 scope:recipient = { is_close_family_of = scope:actor } }
		modifier = { add = -50 scope:actor = { has_trait_xp = { trait = sbs_seed_of_demon value < 100 } } }
	}
}
