on_game_start = {
	effect = {
		set_global_variable = {
			name = sbs_active
			value = yes
		}
	}
}

random_yearly_everyone_pulse = {
	on_actions = {
		sbs_on_action
	}
}

sbs_on_action = {
	trigger = {
		is_adult = yes
		is_ruler = yes
		is_imprisoned = no
		OR = {
			has_trait = sbs_lesser_succubus
			has_trait = sbs_lesser_demon
		}
	}

	effect = {
		if = {
			limit = { has_trait = sbs_lesser_succubus }
			every_relation = {
				type = lover
				root = {
					add_trait_xp = { trait = sbs_lesser_succubus track = corruption value = 1 }
				}
			}
			every_relation = {
				type = soulmate
				root = {
					add_trait_xp = { trait = sbs_lesser_succubus track = loyalty value = 1 }
				}
			}
		}
		if = {
			limit = { has_trait = sbs_lesser_demon }

			every_relation = {
				type = slave
				root = {
					add_trait_xp = { trait = sbs_lesser_demon value = 5 }
				}
			}
		}
	}
}
