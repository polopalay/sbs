sbs_change_character_interaction = {
	icon = propose_best_friendship_interaction
	desc = sbs_change_character_interaction.desc
	common_interaction = no
	auto_accept = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:actor = {
			is_ai = no
			has_trait = slave
		}
		scope:recipient = {
			is_ai = yes
			has_trait = slave
			is_close_family_of = scope:actor
		}
		scope:actor.house = scope:recipient.house
	}

	on_accept = {
		scope:actor = {
			set_player_character = scope:recipient
		}
	}
}
