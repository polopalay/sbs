carnia_slave_training_interaction = {
	category = interaction_category_prison
	desc = carnia_slave_training_interaction.desc
	common_interaction = yes
	cooldown_against_recipient = { years = 1 }
	is_shown = {
		scope:actor = {
			is_adult = yes
		}
		scope:recipient = {
			is_adult = yes
			has_trait = slave
			is_courtier_of = scope:actor
		}
	}
	localization_values = {
		RANSOM_COST = scope:recipient.gold
	}
	send_option = {
		is_shown = { scope:recipient = { should_be_naked_trigger = no } }
		flag = naked
		localization = diplomacy_majesty_4090_no_raiment_modifier
	}
	send_option = {
		is_shown = { scope:recipient.gold > 0 }
		flag = gold
		localization = "RANSOM_GOLD_OPTION"
	}
	send_option = {
		is_shown = { scope:actor.faith != scope:recipient.faith }
		flag = conversion
		localization = "ask_for_conversion_interaction"
	}
	send_option = {
		is_shown = { scope:recipient = { has_trait_rank = { trait = wounded rank < 3 } } }
		flag = torture
		localization = "torture_interaction"
	}
	send_option = {
		is_shown = {  scope:recipient = { is_female = yes is_adult = yes } }
		flag = prostitute
		localization = "carnia_prostitute_interaction"
	}
	send_options_exclusive = yes
	on_accept = {
		scope:recipient = {
			random_list = {
				1 = {
					if = {
						limit = { current_weight < 40 }
						change_current_weight = -10
					}
					else = {
						add_character_modifier = { modifier =  stress_smelling_of_sweat years = 1 }
					}
				}
				0 = {
					modifier = { scope:torture = yes add = 1 }
					change_trait_rank = { trait = wounded rank = 1 }
				}
				0 = {
					modifier = { scope:actor = { is_ai = yes } should_be_naked_trigger = no add = 2 }
					modifier = { scope:naked = yes add = 10 }
					add_character_modifier = diplomacy_majesty_4090_no_raiment_modifier
				}
				0 = {
					modifier = { scope:actor = { is_ai = yes } gold > 50 add = 4 }
					modifier = { scope:gold = yes add = 10 }
					pay_short_term_gold = { target = scope:actor gold = gold }
				}
				0 = {
					modifier = { scope:actor = { is_ai = yes } scope:actor.faith != scope:recipient.faith add = 4 }
					modifier = { scope:conversion = yes add = 10 }
					set_character_faith_with_conversion = scope:actor.faith
				}
				0 = {
					modifier = { scope:actor = { is_ai = yes } is_female = yes is_adult = yes add = 10 }
					modifier = { scope:prostitute = yes add = 10 }
					random_living_character = {
						limit = { is_lowborn = yes is_ruler = no is_female = no is_adult = yes in_diplomatic_range = scope:actor }
						prostitute_effect = { PROSTITUTE = scope:recipient PREGNANCY_CHANCE = 100 }
						pay_short_term_gold = { target = scope:actor gold = tiny_gold_value }
					}
				}
			}
		}
	}
	auto_accept = yes
	ai_targets = { ai_recipients = courtiers }
	ai_potential = { always = yes }
	ai_frequency = 2
	ai_will_do = {
		base = 25
		ai_value_modifier = {
			ai_vengefulness = 1
			ai_greed = 0.5
			ai_compassion = -1
			ai_honor = -0.5
			ai_rationality = -0.1
		}
		modifier = { add = -50 scope:recipient = { has_relation_friend = scope:actor } }
		modifier = { add = -50 scope:recipient = { has_relation_lover = scope:actor } }
		modifier = { add = -100 scope:recipient = { has_relation_soulmate = scope:actor } }
		modifier = { add = -100 scope:recipient = { is_adult = no } }
	}
}
