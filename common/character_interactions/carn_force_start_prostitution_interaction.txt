		scope:actor = {
			is_ruler = yes
			is_adult = yes
		}

			is_adult = yes

				has_character_modifier = carn_working_as_prostitute_1_modifier

			}
			OR = {
				is_courtier_of = scope:actor
				scope:actor = scope:recipient
			}
			has_std_trigger = no
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			any_courtier = {
				OR = {
					has_character_modifier = carn_working_as_prostitute_1_modifier

					has_character_modifier = carn_working_as_prostitute_3_modifier

			}
		}
	}

		is_shown = {

		}
		is_valid = {

			}
		}
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	send_options_exclusive = no

		scope:recipient = {
			carn_start_working_as_prostitute_effect = yes
		}
		scope:actor = {
			if = {
				limit = {
					NOT = { is_liege_or_above_of = scope:recipient }
					scope:recipient != scope:actor
				}
				add_courtier = scope:recipient
			}
			if = {
				limit = {
					scope:hook = yes
					has_hook = scope:recipient
				}
				use_hook = scope:recipient
			}
		}
	}

	auto_accept = {
		OR = {
			scope:recipient = scope:actor
			scope:recipient = { has_trait = slave }
			scope:hook = yes
		}
	}

		base = -20
		opinion_modifier = {

		}

		modifier = { add = -50 scope:recipient = { is_female = yes culture = { has_cultural_pillar = martial_custom_female_only } } }
	}

	ai_frequency = 12

		modifier = { add = 5 scope:actor = { faith = { has_doctrine = carn_doctrine_prostitution_virtuous } } }
		modifier = { add = 10 scope:actor = { has_trait = lustful } }
		modifier = { add = 20 scope:actor = { OR = { has_trait = deviant any_secret = { type = secret_deviant } } } }

		modifier = { add = -20 scope:recipient = { primary_title.tier = tier_duchy is_landless_ruler = no } }
