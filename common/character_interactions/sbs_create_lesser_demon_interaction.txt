sbs_create_lesser_demon_interaction = {
	icon = sbs_create_lesser_demon_interaction
	desc = sbs_create_lesser_demon_interaction.desc
	category = interaction_category_diplomacy
	use_diplomatic_range = yes
	cooldown = { years = 20 }
	common_interaction = yes
	is_shown = {
		NOT = { has_global_variable = seal_the_gates_of_hell }
		scope:recipient = { NOT = { has_trait_with_flag = sbs_demon_trait } }
		scope:actor = { has_trait = sbs_true_demon }
	}
	is_valid = {
		scope:recipient.age >= 16
	}
	localization_values = {
		RANSOM_COST = monumental_gold_value
	}
	send_option = {
		is_shown = { scope:actor != scope:recipient }
		is_valid = { scope:recipient = { gold >= monumental_gold_value } }
		flag = gold
		localization = "RANSOM_GOLD_OPTION"
	}

	send_option = {
		is_shown = { scope:actor != scope:recipient }
		is_valid = { scope:recipient = { has_usable_hook = scope:actor } }
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}

	send_option = {
		is_shown = { scope:actor != scope:recipient }
		is_valid = {
			scope:actor = {
				can_add_hook = {
					target = scope:recipient
					type = favor_hook
				}
			}
		}
		flag = favor
		localization = "RANSOM_FAVOR_OPTION"
	}

	send_option = {
		flag = demand_conversion
		localization = "RELEASE_DEMAND_CONVERSION"
		is_shown = { scope:recipient.faith != scope:actor.faith }
		is_valid = {
			custom_description = {
				text = cannot_demand_religious_head_conversion
				NAND = {
					exists = scope:recipient.faith.religious_head
					scope:recipient.faith.religious_head = scope:recipient
				}
			}
			custom_description = {
				text = cannot_demand_holy_order_master_conversion
				NAND = {
					exists = scope:recipient.faith
					scope:recipient.faith = { any_faith_holy_order = { leader = scope:recipient } }
				}
			}
		}
	}

	send_options_exclusive = no
	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:favor = yes }
				add_hook = { target = scope:recipient type = favor_hook }
			}
		}
		scope:recipient = {
			if = {
				limit = { scope:gold = yes }
				pay_short_term_gold = { target = scope:actor gold = monumental_gold_value }
			}
			if = {
				limit = { scope:hook = yes has_hook = scope:actor }
				use_hook = scope:actor
			}
			if = {
				limit = { scope:demand_conversion = yes }
				set_character_faith = scope:actor.faith
			}
			if = {
				limit = { is_female = yes }
				add_trait = sbs_lesser_succubus
				sbs_mind_break_effect = yes
			}
			else = {
				add_trait = sbs_lesser_demon
				add_character_modifier = { modifier = sbs_buff years = 100 }
			}
			if = {
				limit = { has_trait = saint }
				remove_trait = saint
				give_nickname = nick_the_betrayer
			}
			if = {
				limit = { has_trait = chaste }
				remove_trait = chaste
				add_trait = lustful
			}
			if = {
				limit = { has_trait = immortal }
				remove_trait = immortal
				if = {
					limit = { is_female = yes }
					add_trait_xp = { trait = sbs_lesser_succubus track = corruption value = 100 }
					add_trait_xp = { trait = sbs_lesser_succubus track = power value = 100 }
				}
				else = {
					add_trait_xp = { trait = sbs_lesser_demon value = 100 }
				}
			}
		}
	}
	ai_accept = {
		base = 30
		opinion_modifier = { opinion_target = scope:actor who = scope:recipient multiplier = 0.25 }
		modifier = { add = -40 scope:gold = yes }
		modifier = { add = -15 scope:hook = yes }
		modifier = { add = -15 scope:favor = yes }
		modifier = { add = -20 scope:demand_conversion = yes }
		modifier = { add = 10 scope:actor.faith = scope:recipient.faith }
		modifier = { add = -30 scope:demand_conversion = yes scope:recipient = { has_trait = zealous } }
	}
	ai_frequency = 12
	ai_targets = { ai_recipients = family max = 10 }
	ai_targets = {
		ai_recipients = scripted_relations
		ai_recipients = neighboring_rulers
		ai_recipients = realm_characters
		max = 10
	}
	ai_potential = { always = yes }
	ai_will_do = {
		base = 0
		modifier = { add = 10 scope:recipient = { is_child_of = scope:actor average_of_all_skills_and_prowess >= extremely_high_skill_rating } }
		modifier = { add = 10 scope:recipient.dynasty = scope:actor.dynasty scope:recipient.average_of_all_skills_and_prowess >= extremely_high_skill_rating }
	}
}
