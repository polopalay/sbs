sbs_ask_for_become_succubus_interaction = {
	icon = sbs_ask_for_become_succubus_interaction
	desc = sbs_ask_for_become_succubus_interaction.desc
	category = interaction_category_diplomacy
	use_diplomatic_range = {
		scope:actor = { is_ai = yes }
	}
	common_interaction = yes
	cooldown = { years = 25 }

	is_shown = {
		NOT = { has_global_variable = seal_the_gates_of_hell }
		scope:actor = {
			is_female = yes
			has_trait = sbs_lesser_succubus
		}
		scope:recipient = {
			has_trait = sbs_demon
		}
	}
	is_valid = {
		scope:actor = {
			has_trait_xp = {
				trait = sbs_lesser_succubus
				value >= 100
			}
		}
		scope:recipient = {
			faith.religion = religion:demonic_religion
			NOT = { has_character_modifier = sbs_seal }
		}
	}

	localization_values = {
		RANSOM_COST = major_gold_value
	}
	populate_recipient_list = {
		scope:actor = {
			every_courtier = {
				limit = { NOT = { this = scope:recipient } }
				add_to_list = characters
			}
		}
	}
	send_option = {
		is_valid = {
			scope:actor ?= { gold >= major_gold_value }
		}
		flag = gold
		localization = "RANSOM_GOLD_OPTION"
	}
	send_option = {
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	send_option = {
		is_valid = {
			exists = scope:recipient
			scope:recipient = {
				can_add_hook = {
					target = scope:actor
					type = favor_hook
				}
			}
		}
		flag = favor
		localization = "RANSOM_OWE_FAVOR_OPTION"
	}
	send_option = {
		flag = demand_conversion
		localization = "RELEASE_DEMAND_CONVERSION"
		is_shown = {
			NOT = { scope:actor.faith = scope:recipient.faith }
		}
		is_valid = {
			# Cannot ask Religious Heads to convert
			custom_description = {
				text = cannot_demand_religious_head_conversion
				NOT = {
					AND = {
						exists = scope:actor.faith.religious_head
						scope:actor.faith.religious_head = scope:actor
					}
				}
			}
			# Cannot ask Holy Order Master to convert
			custom_description = {
				text = cannot_demand_holy_order_master_conversion
				NAND = {
					exists = scope:actor.faith
					scope:actor.faith = {
						any_faith_holy_order = { leader = scope:actor }
					}
				}
			}
		}
	}
	send_options_exclusive = no

	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:gold = yes }
				pay_short_term_gold = {
					target = scope:recipient
					gold = major_gold_value
				}
			}
			if = {
				limit = {
					scope:hook = yes
					has_hook = scope:recipient
				}
				use_hook = scope:recipient
			}
			if = {
				limit = {
					scope:demand_conversion = yes
				}
				set_character_faith_with_conversion = scope:recipient.faith
			}
			if = {
				limit = { has_trait = immortal }
				remove_trait = immortal
			}
			remove_trait = sbs_lesser_succubus
			add_trait = sbs_succubus
			if = {
				limit = { scope:secondary_recipient = { is_close_family_of = scope:actor } }
				add_trait_xp = { trait = sbs_succubus track = corruption value = 50 }
			}
		}
		scope:recipient = {
			if = {
				limit = { scope:favor = yes }
				add_hook = {
					target = scope:actor
					type = fabrication_hook
				}
			}
			add_character_modifier = { modifier = sbs_seal years = 25 }
		}
		scope:secondary_recipient = {
			death = {
				death_reason = death_execution
				killer = scope:actor
			}
		}
	}

	ai_accept = {
		base = -200
		opinion_modifier = {
			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.25
		}
		opinion_modifier = {
			opinion_target = scope:secondary_recipient
			who = scope:recipient
			multiplier = -0.1
		}
		modifier = { add = 20 scope:hook = yes }
		modifier = { add = 20 scope:gold = yes }
		modifier = { add = 20 scope:favor = yes }
		modifier = { add = 10 scope:actor.faith = scope:recipient.faith }
		modifier = { add = 10 scope:actor.culture = scope:recipient.culture }
		modifier = { add = 40 scope:demand_conversion = yes scope:actor.primary_title.tier = tier_kingdom }
		modifier = { add = 80 scope:demand_conversion = yes scope:actor.primary_title.tier = tier_empire }
		modifier = { add = 100 scope:demand_conversion = yes scope:actor.primary_title.tier = tier_hegemony }
		modifier = { add = 25 scope:actor = { OR = { has_trait = deviant any_secret = { type = secret_deviant is_known_by = scope:recipient } } } } 
		modifier = { add = 50 scope:actor = { OR = { has_trait = witch any_secret = { type = secret_witch is_known_by = scope:recipient } } } }
		modifier = { add = 50  scope:actor = { has_relation_soulmate = scope:recipient } }
		modifier = { add = 50  scope:actor = { has_relation_lover = scope:recipient } }
		modifier = { add = 50  scope:actor = { has_relation_friend = scope:recipient } }
		modifier = { add = 100  scope:actor = { has_relation_best_friend = scope:recipient } }
		modifier = { add = 20 scope:actor = { is_extended_family_of = scope:recipient } }
		modifier = { add = 40 scope:actor = { is_close_family_of = scope:recipient } }
		modifier = { add = 80 scope:actor = { is_child_of = scope:recipient } }
	}
	ai_potential = { always = yes }
	ai_frequency = 12
	ai_targets = {
		ai_recipients = head_of_faith
	}
	ai_targets = {
		ai_recipients = family
		max = 10
	}
	ai_targets = {
		ai_recipients = scripted_relations
		ai_recipients = neighboring_rulers
		max = 10
	}

	ai_will_do = {
		base = 100
	}
}
