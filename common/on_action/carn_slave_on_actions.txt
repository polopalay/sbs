on_death_relation_slave = { # fired for the slaveowner when they die
	effect = {
		# inheritance of slaves
		if = {
			limit = { exists = primary_heir }
			primary_heir = {
				set_relation_slave = scope:target
				scope:target = {
					set_relation_slave_owner = prev
				}
				add_courtier = scope:target
			}
		}
	}
}
