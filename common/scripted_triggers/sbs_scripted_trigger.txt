is_valid_construction_alchemy_trigger = {
	OR = {
		any_sub_realm_county = { any_county_province = { has_building_or_higher = alchemy1 } }
		any_realm_county = { any_county_province = { has_building_or_higher = alchemy2 } }
		any_county_in_region = { any_county_province = { has_building_or_higher = alchemy3 } }
	}
}
