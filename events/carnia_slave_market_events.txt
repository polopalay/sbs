		any_living_character = {
			is_alive = yes

		}

	immediate = {
		random_living_character = {
			limit = {
				is_alive = yes

				NOT = {
					any_close_family_member = { is_ruler = yes }
				}
			}

		}
	}

	left_portrait = { character = scope:carnia_slave_noble animation = sadness }
	right_portrait = { character = root animation = personality_greedy }

	option = {
