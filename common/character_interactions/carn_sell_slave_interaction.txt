carn_sell_slave_interaction = {
	category = interaction_category_prison
	desc = carn_sell_slave_interaction.desc
	common_interaction = yes
	needs_recipient_to_open = yes
	cooldown_against_recipient = { years = 3 }
	send_options_exclusive = no
	is_shown = {
		scope:actor = {
			is_ruler = yes
			any_courtier = { has_trait = slave has_relation_slave_owner = scope:actor }
		}
		scope:recipient = { is_ruler = yes }
		scope:actor != scope:recipient
	}
	populate_recipient_list = {
		scope:actor = {
			every_courtier = {
				limit = { has_trait = slave has_relation_slave_owner = scope:actor NOT = { has_character_flag = carnia_slave_sale_pending } }
				add_to_list = characters
			}
		}
	}
	localization_values = {
		EXTORTIONATE_RANSOM_COST = scope:secondary_recipient.carn_slave_price_value
		RANSOM_COST = scope:secondary_recipient.carn_slave_price_value
	}
	send_option = {
		is_shown = {
			scope:actor = { dynasty ?= { has_dynasty_perk = fp1_pillage_legacy_3 } }
			scope:recipient ?= {
				gold >= {
					value = scope:secondary_recipient.carn_slave_price_value
					multiply = 2
				}
			}
		}
		flag = extortionate_gold
		localization = "RANSOM_EXTORTIONATE_GOLD_OPTION"
	}
	send_option = {
		is_shown = {
			scope:actor = {
				OR = {
					is_lowborn = yes
					NOT = { dynasty = { has_dynasty_perk = fp1_pillage_legacy_3 } }
				}
			}
			scope:recipient ?= {
				gold >= {
					value = scope:secondary_recipient.carn_slave_price_value
					multiply = 1.5
				}
			}
		}
		flag = gold
		localization = "RANSOM_GOLD_OPTION"
	}
	send_option = {
		is_valid = { scope:recipient = { has_usable_hook = scope:actor } }
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	send_option = {
		is_valid = {
			scope:actor = {
				can_add_hook = { target = scope:recipient type = favor_hook }
			}
		}
		flag = favor
		localization = "RANSOM_FAVOR_OPTION"
	}
	on_send = {
		scope:secondary_recipient = {
			add_character_flag = carnia_slave_sale_pending
		}
	}
	on_accept = {
		scope:actor = { save_scope_as = seller }
		scope:recipient = { save_scope_as = buyer }
		scope:secondary_recipient = {
			save_scope_as = slave
			remove_character_flag = carnia_slave_sale_pending
		}
		carn_buy_slave_effect = { SLAVE_PRICE = scope:secondary_recipient.carn_slave_price_value }
	}
	on_decline = {
		scope:secondary_recipient = {
			remove_character_flag = carnia_slave_sale_pending
		}
	}
	ai_accept = {
		base = 5
		opinion_modifier = { opinion_target = scope:actor multiplier = 0.1 }
		opinion_modifier = { opinion_target = scope:secondary_recipient multiplier = 0.2 }
		modifier = { add = 5 scope:secondary_recipient = { has_trait = beauty_good_1 } }
		modifier = { add = 10 scope:secondary_recipient = { has_trait = beauty_good_2 } }
		modifier = { add = 20 scope:secondary_recipient = { has_trait = beauty_good_3 } }
		modifier = { add = -5 scope:secondary_recipient = { has_trait = beauty_bad_1 } }
		modifier = { add = -10 scope:secondary_recipient = { has_trait = beauty_bad_2 } }
		modifier = { add = -20 scope:secondary_recipient = { has_trait = beauty_bad_3 } }
		modifier = { add = 5 scope:secondary_recipient = { has_trait = intellect_good_1 } }
		modifier = { add = 8 scope:secondary_recipient = { has_trait = intellect_good_2 } }
		modifier = { add = 10 scope:secondary_recipient = { has_trait = intellect_good_3 } }
		modifier = { add = -5 scope:secondary_recipient = { has_trait = intellect_bad_1 } }
		modifier = { add = -8 scope:secondary_recipient = { has_trait = intellect_bad_2 } }
		modifier = { add = -10 scope:secondary_recipient = { has_trait = intellect_bad_3 } }
		modifier = { add = 5 scope:secondary_recipient = { has_trait = physique_good_1 } }
		modifier = { add = 8 scope:secondary_recipient = { has_trait = physique_good_2 } }
		modifier = { add = 10 scope:secondary_recipient = { has_trait = physique_good_3 } }
		modifier = { add = -5 scope:secondary_recipient = { has_trait = physique_bad_1 } }
		modifier = { add = -8 scope:secondary_recipient = { has_trait = physique_bad_2 } }
		modifier = { add = -10 scope:secondary_recipient = { has_trait = physique_bad_3 } }
		modifier = { add = 5 scope:secondary_recipient = { has_trait = pure_blooded } }
		modifier = { add = 5 scope:secondary_recipient = { has_trait = strong } }
		modifier = { add = 5 scope:secondary_recipient = { has_trait = shrewd } }
		modifier = { add = -5 scope:secondary_recipient = { has_trait = clubfooted } }
		modifier = { add = -5 scope:secondary_recipient = { has_trait = hunchbacked } }
		modifier = { add = -5 scope:secondary_recipient = { has_trait = lisping } }
		modifier = { add = -5 scope:secondary_recipient = { has_trait = stuttering } }
		modifier = { add = -5 scope:secondary_recipient = { has_trait = dwarf } }
		modifier = { add = -5 scope:secondary_recipient = { has_trait = giant } }
		modifier = { add = -5 scope:secondary_recipient = { has_trait = weak } }
		modifier = { add = -5 scope:secondary_recipient = { has_trait = dull } }
		modifier = { add = -15 scope:extortionate_gold = yes }
		modifier = { add = -10 scope:gold = yes }
		modifier = { add = -10 scope:hook = yes }
		modifier = { add = -10 scope:favor = yes }
		modifier = { add = 10 desc = carn_secondary_recipient_is_my_friend_reason scope:secondary_recipient = { has_relation_friend = scope:recipient } }
		modifier = { add = 20 desc = carn_secondary_recipient_is_my_lover_reason scope:secondary_recipient = { has_relation_lover = scope:recipient } }
		modifier = { add = 10 desc = carn_secondary_recipient_is_my_crush_reason scope:secondary_recipient = { has_relation_crush = scope:recipient } }
		modifier = { add = 20 desc = carn_secondary_recipient_is_my_soulmate_reason scope:secondary_recipient = { has_relation_soulmate = scope:recipient } }
	}
	ai_frequency = 12
	ai_potential = { always = yes }
	ai_targets = { ai_recipients = liege }
	ai_targets = {
		ai_recipients = vassals
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		max = 5
	}
	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = nearby_domicile_owners
		max = 5
	}
	ai_targets = { ai_recipients = realm_characters max = 10 }
	ai_will_do = {
		base = 0
		modifier = {
			factor = 0
			NOR = {
				scope:extortionate_gold = yes
				scope:gold = yes
				scope:hook = yes
				scope:favor = yes
			}
		}
		modifier = { add = 100 scope:actor = { any_relation = { type = slave count > 5 } } }
	}
}
