sbs_seal_the_devil_interaction = {
	desc = sbs_seal_the_devil_interaction.desc
	category = interaction_category_hostile
	use_diplomatic_range = no
	common_interaction = yes
	is_shown = {
		scope:actor = {
			is_ruler = yes
			faith.religion = religion:christianity_religion
			NOT = { has_trait_with_flag = sbs_demon_trait }
		}
		scope:recipient = {
			has_trait_with_flag = sbs_demon_trait
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = {
			OR = {
				AND = {
					is_landed = yes
					capital_province.faith = faith
				}
				AND = {
					is_landed = no
					OR = {
						province:2065.religion = religion:christianity_religion
						province:5965.religion = religion:christianity_religion
						province:482.religion = religion:christianity_religion
						province:2575.religion = religion:christianity_religion
						province:4582.religion = religion:christianity_religion
						province:6038.religion = religion:christianity_religion
					}
				}
			}
		}
		scope:recipient = {
			OR = {
				AND = {
					has_trait = sbs_true_demon
					is_imprisoned_by = scope:actor
				}
				AND = {
					NOT = { has_trait = sbs_true_demon }
					OR = {
						AND = {
							faith.religious_head = scope:recipient
							faith = faith:nature
							is_lowborn = yes
						}
						AND = {
							faith.religious_head = scope:recipient
							faith = faith:pleasure
							is_lowborn = yes
						}
						AND = {
							faith.religious_head = scope:recipient
							faith = faith:violence
							is_lowborn = yes
						}
						AND = {
							faith.religious_head = scope:recipient
							faith = faith:greedy
							is_lowborn = yes
						}
						is_imprisoned_by = scope:actor
					}
					OR = {
						custom_description = {
							text = NOT_I_AM_HEAD_OF_FAITH_TRIGGER
							NOT = { faith.religious_head = scope:recipient }
						}
						AND = {
							faith.religious_head = scope:recipient
							province:2065.religion = religion:christianity_religion
							faith = faith:nature
							custom_description = {
								text = nature_isn_seal
								NOT = { has_global_variable = seal_nature }
							}
						}
						AND = {
							faith.religious_head = scope:recipient
							province:5965.religion = religion:christianity_religion
							faith = faith:pleasure
							custom_description = {
								text = pleasure_isn_seal
								NOT = { has_global_variable = seal_pleasure }
							}
						}
						AND = {
							faith.religious_head = scope:recipient
							province:482.religion = religion:christianity_religion
							faith = faith:violence
							custom_description = {
								text = violence_isn_seal
								NOT = { has_global_variable = seal_violence }
							}
						}
						AND = {
							faith.religious_head = scope:recipient
							province:2575.religion = religion:christianity_religion
							faith = faith:greedy
							custom_description = {
								text = greedy_isn_seal
								NOT = { has_global_variable = seal_greedy }
							}
						}
					}
				}
			}
		}

	}
	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:recipient = { NOT = { has_trait = sbs_true_demon } } }
				if = {
					limit = { NOT = { has_trait = saint } }
					add_trait = saint
					give_nickname = nick_the_saint
				}
				else_if = {
					limit = { NOT = { has_trait = immortal } }
					add_trait = immortal
				}
				else = {
					add_piety = 10000
				}
			}
		}
		scope:recipient = {
			if = {
				limit = { has_trait = sbs_true_demon }
				death = { death_reason = death_execution killer = scope:actor }
				scope:actor = {
					if = { limit = { NOT = { has_trait = faith_warrior } } add_trait = faith_warrior }
					if = { limit = { NOT = { has_trait = saint } } add_trait = saint }
					if = { limit = { NOT = { has_trait = crusader_king } } add_trait = crusader_king }
					if = { limit = { NOT = { has_trait = conqueror } } add_trait = conqueror }
					if = { limit = { NOT = { has_trait = immortal } } add_trait = immortal }
					set_immortal_age = 18

					add_diplomacy_skill = 12
					add_martial_skill = 12
					add_stewardship_skill = 12
					add_intrigue_skill = 12
					add_learning_skill = 12
					add_prowess_skill = 12

					add_piety_level = 6
					add_prestige_level = 6
					dynasty ?= { add_dynasty_prestige = 100000 }
					give_nickname = nick_the_saint
				}

				if = {
					limit = { faith = faith:nature }
					set_global_variable = { name = seal_nature value = yes }
				}
				else_if = {
					limit = { faith = faith:pleasure }
					set_global_variable = { name = seal_pleasure value = yes }
				}
				else_if = {
					limit = { faith = faith:violence }
					set_global_variable = { name = seal_violence value = yes }
				}
				else_if = {
					limit = { faith = faith:greedy }
					set_global_variable = { name = seal_greedy value = yes }
				}
			}
			else_if = {
				limit = { faith.religious_head = scope:recipient faith = faith:nature }
				death = { death_reason = death_execution killer = scope:actor }
				scope:actor = {
					set_trait_rank = { trait = education_learning rank = 5 }
				}
				custom_description = {
					text = seal_nature
					set_global_variable = { name = seal_nature value = yes }
				}
			}
			else_if = {
				limit = { faith.religious_head = scope:recipient faith = faith:pleasure }
				death = { death_reason = death_execution killer = scope:actor }
				scope:actor = {
					set_trait_rank = { trait = education_intrigue rank = 5 }
				}
				custom_description = {
					text = "seal_pleasure"
					set_global_variable = { name = seal_pleasure value = yes }
				}
			}
			else_if = {
				limit = { faith.religious_head = scope:recipient faith = faith:violence }
				death = { death_reason = death_execution killer = scope:actor }
				scope:actor = {
					add_trait = conqueror
				}
				custom_description = {
					text = "seal_violence"
					set_global_variable = { name = seal_violence value = yes }
				}
			}
			else_if = {
				limit = { faith.religious_head = scope:recipient faith = faith:greedy }
				death = { death_reason = death_execution killer = scope:actor }
				scope:actor = {
					set_trait_rank = { trait = education_stewardship rank = 5 }
				}
				custom_description = {
					text = "seal_greedy"
					set_global_variable = { name = seal_greedy value = yes }
				}
			}
			else_if = {
				limit = { any_parent = { has_trait_with_flag = sbs_demon_trait } }
				death = { death_reason = death_execution killer = scope:actor }
				if = {
					limit = { has_trait = education_learning }
					scope:actor = {
						if = {
							limit = { has_trait = education_learning }
							change_trait_rank = { trait = education_learning rank = 1 }
						}
						else = {
							add_trait = education_learning_1	
						}
					}
				}
				else_if = {
					limit = { has_trait = education_intrigue }
					scope:actor = {
						if = {
							limit = { has_trait = education_intrigue }
							change_trait_rank = { trait = education_intrigue rank = 1 }
						}
						else = {
							add_trait = education_intrigue_1
						}
					}
				}
				else_if = {
					limit = { has_trait = education_martial }
					scope:actor = {
						if = {
							limit = { has_trait = education_martial }
							change_trait_rank = { trait = education_martial rank = 1 }
						}
						else = {
							add_trait = education_martial_1
						}
					}
				}
				else_if = {
					limit = { has_trait = education_diplomacy }
					scope:actor = {
						if = {
							limit = { has_trait = education_diplomacy }
							change_trait_rank = { trait = education_diplomacy rank = 1 }
						}
						else = {
							add_trait = education_diplomacy_1
						}
					}
				}
				else_if = {
					limit = { has_trait = education_stewardship }
					scope:actor = {
						if = {
							limit = { has_trait = education_stewardship }
							change_trait_rank = { trait = education_stewardship rank = 1 }
						}
						else = {
							add_trait = education_stewardship_1
						}
					}
				}
			}
		}
	}
	auto_accept = yes
	ai_frequency = 6
	ai_targets = {
		ai_recipients = prisoners
	}
	ai_potential = { always = yes }
	ai_will_do = {
		base = 50
		modifier = { add = -80 scope:actor = { ai_zeal < very_low_positive_ai_value } }
		modifier = { add = -40 scope:actor = { ai_zeal < low_positive_ai_value } }
		modifier = { add = -20 scope:actor = { ai_zeal < medium_positive_ai_value } }
		modifier = { add = 20 scope:actor = { ai_zeal > high_positive_ai_value } }
		modifier = { add = 40 scope:actor = { ai_zeal > very_high_positive_ai_value } }
		modifier = { add = 80 scope:actor = { ai_zeal > dominant_positive_ai_value } }
	}
}
