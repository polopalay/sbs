carn_force_stop_prostitution_interaction = {
	desc = carn_force_stop_prostitution_interaction.desc
	category = interaction_category_prison
	use_diplomatic_range = yes
	common_interaction = yes
	cooldown_against_recipient = { years = 2 }

	is_shown = {
		scope:actor = {
			is_ruler = yes
			is_adult = yes
		}
		scope:recipient = {
			is_adult = yes
			OR = {
				has_character_modifier = carn_working_as_prostitute_1_modifier
				has_character_modifier = carn_working_as_prostitute_2_modifier
				has_character_modifier = carn_working_as_prostitute_3_modifier
			}
			OR = {
				scope:actor = scope:recipient
				AND = {
					is_courtier_of = scope:actor
					OR = {
						any_secret = { type = secret_prostitute is_known_by = scope:actor }
						has_trait = lifestyle_prostitute
					}
				}
			}
		}
	}

	send_option = {
		is_shown = {
			NOT = { scope:actor = scope:recipient }
		}
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	send_option = {
		is_shown = {
			NOT = { scope:actor = scope:recipient }
			scope:recipient.liege = scope:actor
		}
		is_valid = {
			scope:actor = {
				has_imprisonment_reason = scope:recipient
			}
		}
		flag = forgive
		localization = forgive_interaction
	}
	send_options_exclusive = no

	on_accept = {
		scope:recipient = {
			remove_character_modifier = carn_working_as_prostitute_1_modifier
			remove_character_modifier = carn_working_as_prostitute_2_modifier
			remove_character_modifier = carn_working_as_prostitute_3_modifier
			if = {
				limit = {
					has_trait = lifestyle_prostitute
					trait_is_shunned_or_criminal_in_faith_or_liege_trigger = { TRAIT = lifestyle_prostitute }
				}
				remove_trait = lifestyle_prostitute
				add_secret = { type = secret_prostitute }
			}
		}
		scope:actor = {
			if = {
				limit = {
					scope:forgive = yes
					has_imprisonment_reason = scope:recipient
				}
				consume_imprisonment_reasons = scope:recipient
			}
			if = {
				limit = {
					scope:hook = yes
					has_hook = scope:recipient
				}
				use_hook = scope:recipient
			}
		}
	}

	auto_accept = {
		OR = {
			scope:recipient = scope:actor
			scope:recipient = { has_trait = slave }
			scope:hook = yes
			scope:forgive = yes
		}
	}
	ai_accept = {
		base = 0
		opinion_modifier = {
			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.2
		}
		modifier = { add = 50 scope:forgive = yes }
		modifier = { add = 50 scope:recipient = { has_trait = chaste } }
		modifier = { add = 50 scope:recipient.faith = { has_doctrine_parameter = carn_prostitution_shunned } }
		modifier = { add = 100 scope:recipient.faith = { has_doctrine_parameter = carn_prostitution_illegal } }
		modifier = { add = -50 scope:recipient = { faith = { has_doctrine = tenet_polyamory } } }
		modifier = { add = -50 scope:recipient = { faith = { has_doctrine = carn_doctrine_prostitution_virtuous } } }
		modifier = { add = -10 scope:recipient = { OR = { has_trait = deviant any_secret = { type = secret_deviant } } } }
		modifier = { add = -25 scope:recipient = { has_trait = lustful } }
		modifier = { add = 50 scope:recipient = { is_female = no faith = { has_doctrine = doctrine_gender_male_dominated } } }
		modifier = { add = 50 scope:recipient = { is_female = yes faith = { has_doctrine = doctrine_gender_female_dominated } } }
	}

	ai_frequency = 12
	ai_targets = {
		ai_recipients = self
		ai_recipients = courtiers
	}
	ai_potential = { always = yes }
	ai_will_do = {
		base = 0
		modifier = { add = -25 scope:actor = { has_trait = lustful } }
		modifier = { add = -25 scope:actor = { OR = { has_trait = deviant any_secret = { type = secret_deviant } } } }
		modifier = { add = -25 scope:actor = { faith = { has_doctrine = tenet_polyamory } } }
		modifier = { add = 20 scope:recipient = { age > 25 } }
		modifier = { add = 10 scope:recipient = { primary_title.tier = tier_county is_landed_or_landless_administrative = yes } }
		modifier = { add = 20 scope:recipient = { primary_title.tier = tier_duchy is_landed_or_landless_administrative = yes } }
		modifier = { add = 30 scope:recipient = { primary_title.tier = tier_kingdom is_landed_or_landless_administrative = yes } }
		modifier = { add = 40 scope:recipient = { primary_title.tier = tier_empire is_landed_or_landless_administrative = yes} }
		modifier = { add = 50 scope:recipient = { primary_title.tier = tier_hegemony is_landed_or_landless_administrative = yes} }
		modifier = { add = 40 scope:actor = { has_trait = chaste } }
		modifier = { add = 50 scope:recipient = { is_female = no faith = { has_doctrine = doctrine_gender_male_dominated } } }
		modifier = { add = 50 scope:recipient = { is_female = yes faith = { has_doctrine = doctrine_gender_female_dominated } } }
	}
}
