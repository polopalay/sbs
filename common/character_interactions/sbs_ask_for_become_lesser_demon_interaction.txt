sbs_ask_for_become_lesser_demon_interaction = {
	icon = sbs_ask_for_become_lesser_demon_interaction
	desc = sbs_ask_for_become_lesser_demon_interaction.desc
	category = interaction_category_diplomacy
	use_diplomatic_range = {
		scope:actor = { is_ai = yes }
	}
	common_interaction = yes
	cooldown = { years = 10 }

	is_shown = {
		NOT = { has_global_variable = seal_the_gates_of_hell }
		scope:actor = {
			NOT = { has_trait_with_flag = sbs_demon_trait }
		}
		scope:recipient = {
			OR = {
				has_trait = sbs_demon
				AND = {
					has_trait = sbs_succubus
					is_valid_construction_alchemy3_trigger = yes
				}
			}
		}
	}

	is_valid = {
		scope:actor = {
			OR = {
				has_trait = deviant
				any_secret = { type = secret_deviant }
			}
			ON = {
				has_trait = witch
				any_secret = { type = secret_witch }
			}
		}
	}

	localization_values = {
		RANSOM_COST = massive_gold_value
	}
	send_option = {
		is_valid = { scope:actor ?= { gold >= massive_gold_value } }
		flag = gold
		localization = "RANSOM_GOLD_OPTION"
	}
	send_option = {
		is_valid = { scope:actor = { has_usable_hook = scope:recipient } }
		flag = hook
		localization = GENERIC_SPEND_A_HOOK
	}
	send_option = {
		is_valid = {
			exists = scope:recipient
			scope:recipient = {
				can_add_hook = {
					target = scope:actor
					type = favor_hook
				}
			}
		}
		flag = favor
		localization = "RANSOM_OWE_FAVOR_OPTION"
	}
	send_option = {
		flag = demand_conversion
		localization = "RELEASE_DEMAND_CONVERSION"
		is_shown = { NOT = { scope:actor.faith = scope:recipient.faith } }
		is_valid = {
			# Cannot ask Religious Heads to convert
			custom_description = {
				text = cannot_demand_religious_head_conversion
				NOT = {
					AND = {
						exists = scope:actor.faith.religious_head
						scope:actor.faith.religious_head = scope:actor
					}
				}
			}
			# Cannot ask Holy Order Master to convert
			custom_description = {
				text = cannot_demand_holy_order_master_conversion
				NAND = {
					exists = scope:actor.faith
					scope:actor.faith = {
						any_faith_holy_order = { leader = scope:actor }
					}
				}
			}
		}
	}
	send_options_exclusive = no

	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:gold = yes }
				pay_short_term_gold = {
					target = scope:recipient
					gold = major_gold_value
				}
			}
			if = {
				limit = {
					scope:hook = yes
					has_hook = scope:recipient
				}
				use_hook = scope:recipient
			}
			if = {
				limit = { scope:demand_conversion = yes }
				set_character_faith_with_conversion = scope:recipient.faith
			}
			if = {
				limit = { is_female = yes }
				add_trait = sbs_lesser_succubus
			}
			else = {
				add_trait = sbs_seed_of_demon
			}
			if = {
				limit = { has_trait = saint }
				remove_trait = saint
				set_character_faith_with_conversion = scope:recipient.faith
				give_nickname = nick_the_betrayer
			}
			if = {
				limit = { has_trait = immortal }
				remove_trait = immortal
				add_trait_xp = { trait = sbs_lesser_succubus value = 10 }
			}
		}
		scope:recipient = {
			if = {
				limit = { scope:favor = yes }
				add_hook = {
					target = scope:actor
					type = fabrication_hook
				}
			}
		}
	}
	ai_accept = {
		base = -25
		opinion_modifier = {
			opinion_target = scope:actor
			who = scope:recipient
			multiplier = 0.25
		}
		modifier = { add = 5 scope:hook = yes }
		modifier = { add = 5 scope:gold = yes }
		modifier = { add = 5 scope:favor = yes }
		modifier = { add = 10 scope:actor.faith = scope:recipient.faith }
		modifier = { add = 10 scope:actor.culture = scope:recipient.culture }
		modifier = { add = 10 scope:demand_conversion = yes scope:actor = { primary_title.tier = tier_kingdom } }
		modifier = { add = 20 scope:demand_conversion = yes scope:actor = { primary_title.tier = tier_empire } }
		modifier = { add = 30 scope:demand_conversion = yes scope:recipient = { primary_title.tier = tier_hegemony } }
		modifier = { add = 50 scope:actor = { has_trait = saint } }
	}

	ai_potential = { always = yes }
	ai_frequency = 12
	ai_targets = {
		ai_recipients = family
		ai_recipients = scripted_relations
		ai_recipients = head_of_faith
		ai_recipients = liege
		ai_recipients = vassals
		ai_recipients = peer_vassals
		ai_recipients = courtiers
		ai_recipients = neighboring_rulers
		ai_recipients = nearby_domicile_owners
		max = 5
	}

	ai_will_do = {
		base = 100
	}
}
